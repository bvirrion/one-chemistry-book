\documentclass[11pt,a4paper,twoside,openany]{book}

\newcommand{\booklang}{ar}
\usepackage{styles/onechemistry}

% Brand the running header: rosette + ONE-COURSE.COM (linking to the site)
% above the chapter mark and the book title, on even pages; page number at the
% foot. See \ombrandheader in styles/onechemistry.sty.
\ombrandheader

% Print, at the end of every exercise and weekend problem, a hyperlinked
% "Solution p. N" pointer to its solution in the end-of-book appendix.
\omsolutionlinks

% Unbreakable in-text figures (omfigure) that do not fit make a twoside book
% stretch the glue above them into large mid-page gaps; pages ending a little
% short are the standard remedy, as in Book 3 (pending a series-wide decision
% in styles/onechemistry.sty).
\raggedbottom

% Arabic only. babel's bidi=basic boxes every inline $...$, so interword glue
% barely stretches and a formula-dense paragraph can end up with no legal
% break point at all (see arabic_style_card.md; it belongs in the ENTRY file,
% never in a chapter body, which must mirror the English canon line for line).
\setlength{\emergencystretch}{3em}

% Version and date default to empty / today, but a release build supplies a
% generated version.tex that overrides them (see .github/workflows/release.yml).
