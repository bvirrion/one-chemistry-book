\begin{solution}{exo:g9:elements-universe-earth:5}

\begin{solution}{exo:g9:elements-universe-earth:6}
They are very light gases: they escaped from the young Earth into space
(and helium forms no compounds that could have kept it in the rocks).
\end{solution}

\begin{solution}{exo:g9:elements-universe-earth:7}

\begin{solution}{exo:g9:elements-universe-earth:8}

\begin{solution}{exo:g9:elements-universe-earth:9}

\end{solution}

\begin{solution}{exo:g9:elements-universe-earth:10}
Oxygen and calcium (and, beyond the eight main elements of the crust,
hydrogen, carbon and phosphorus are present in the crust in small

\end{solution}

\begin{solution}{exo:g9:elements-universe-earth:11}
$40 \times 1.67 \times 10^{-27} = \qty{6.68e-26}{kg}$;
$1.06 / 6.68 \times 10^{-26} \approx 1.6 \times 10^{25}$ calcium \omterm{def:g7:atoms-and-molecules:atom}{atoms}.
\end{solution}

\begin{solution}{exo:g9:elements-universe-earth:12}
\omterm{def:g7:atoms-and-molecules:atom}{Atoms} are neither created nor destroyed by \omterm{def:g5:irreversible-changes:chemical-change}{chemical changes}: they only
pass from one compound to another (rock, gas, sugar, living body). The
carbon \omterm{def:g7:atoms-and-molecules:atom}{atoms} of the Earth are used again and again.
\end{solution}

\begin{solution}{pb:g9:elements-universe-earth:1}
\textbf{1.} $0.61 \times 70 = \qty{42.7}{kg}$.

\textbf{2.} Carbon $0.23 \times 70 = \qty{16.1}{kg}$; hydrogen
$0.10 \times 70 = \qty{7.0}{kg}$.

\textbf{3.} $61 + 23 + 10 = \qty{94}{\percent}$.

\textbf{4.} Water, \ce{H2O}.

\textbf{5.} $16 \times 1.67 \times 10^{-27} = \qty{2.67e-26}{kg}$.

\textbf{6.} Carbon: $12 \times 1.67 \times 10^{-27} = \qty{2.00e-26}{kg}$;
hydrogen: \qty{1.67e-27}{kg}.

\textbf{7.} 16.

\textbf{8.} An \omterm{def:g9:inside-the-atom:nucleus}{electron} is about 1836 times lighter than a \omterm{def:g9:inside-the-atom:proton}{nucleon}: the
\omterm{def:g9:inside-the-atom:nucleus}{electrons} are less than a thousandth of the mass.

\textbf{9.} $7.0 / 1.67 \times 10^{-27} \approx 4.2 \times 10^{27}$
hydrogen \omterm{def:g7:atoms-and-molecules:atom}{atoms}.

\textbf{10.} $16.1 / 2.00 \times 10^{-26} \approx 8.0 \times 10^{26}$
carbon \omterm{def:g7:atoms-and-molecules:atom}{atoms}.

\textbf{11.} $42.7 / 2.67 \times 10^{-26} \approx 1.60 \times 10^{27}$
oxygen \omterm{def:g7:atoms-and-molecules:atom}{atoms}.

\textbf{12.} Our count, $1.60 \times 10^{27}$, agrees with the estimate
$1.61 \times 10^{27}$: a \qty{70}{kg} body holds about $1.6 \times
10^{27}$ oxygen \omterm{def:g7:atoms-and-molecules:atom}{atoms}.
\end{solution}
