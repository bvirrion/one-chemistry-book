\section*{Chapitre \ref{ch:g9:acids-bases-ph} --- Acides, bases et pH}

\begin{solution}{exo:g9:acids-bases-ph:1}
\omterm{def:g9:acids-bases-ph:ph}{pH} 3~: acide. \omterm{def:g9:acids-bases-ph:ph}{pH} 7~: neutre. \omterm{def:g9:acids-bases-ph:ph}{pH} 11~: basique.
\end{solution}

\begin{solution}{exo:g9:acids-bases-ph:2}
\omterm{def:g9:acids-bases-ph:acidic}{Solution acide}~: les \omterm{def:g9:ions:ion}{ions} \ce{H+}. \omterm{def:g9:acids-bases-ph:acidic}{Solution basique}~: les \omterm{def:g9:ions:ion}{ions} \ce{OH-}.
\end{solution}

\begin{solution}{exo:g9:acids-bases-ph:3}
Le jus de citron vert (2), le café (5), l'eau pure (7), l'eau de mer
(8.1), l'ammoniaque ménagère (12).
\end{solution}

\begin{solution}{exo:g9:acids-bases-ph:4}
On dépose sur une bande posée dans une coupelle propre et sèche une
goutte de la solution prélevée avec un agitateur en verre propre~; on
compare aussitôt la couleur à l'échelle.
\end{solution}

\begin{solution}{exo:g9:acids-bases-ph:5}
Environ 5 après une dilution au dixième~; environ 6 après une dilution
au centième.
\end{solution}

\begin{solution}{exo:g9:acids-bases-ph:6}
Non. Diluer rapproche le \omterm{def:g9:acids-bases-ph:ph}{pH} de 7, mais ne le fait jamais dépasser~:
l'eau elle-même est neutre, et en ajouter ne peut pas rendre les \omterm{def:g9:ions:ion}{ions}
\ce{OH-} plus nombreux que les \omterm{def:g9:ions:ion}{ions} \ce{H+}.
\end{solution}

\begin{solution}{exo:g9:acids-bases-ph:7}
Le \omterm{def:g6:pure-substances-and-mixtures:mixture}{mélange} dégage de la chaleur. De l'eau versée sur un acide concentré
pourrait bouillir aussitôt et projeter de l'acide~; de l'acide versé
lentement dans l'eau se répartit aussitôt dans un grand volume.
\end{solution}

\begin{solution}{exo:g9:acids-bases-ph:8}
Corrosif~: brûle la peau et les yeux, attaque les métaux. Porter des
lunettes et des gants~; rincer toute éclaboussure à grande eau.
\end{solution}

\begin{solution}{exo:g9:acids-bases-ph:9}
Trois dilutions (de 1 à 2, de 2 à 3, de 3 à 4)~: $10 \times 10 \times 10 =
1000$ fois.
\end{solution}

\begin{solution}{exo:g9:acids-bases-ph:10}
Le lait de magnésie est basique (\omterm{def:g9:acids-bases-ph:ph}{pH} d'environ 10.5)~: il réagit avec une
partie de l'excès d'acide de l'estomac.
\end{solution}

\begin{solution}{exo:g9:acids-bases-ph:11}
Non~: avec un \omterm{def:g9:acids-bases-ph:ph}{pH} de 8.1, il est encore légèrement basique. Il évolue vers
un \omterm{def:g9:acids-bases-ph:ph}{pH} plus bas, moins basique, parce que du dioxyde de carbone de l'\omterm{def:g4:air-a-mixture-of-gases:air}{air}
s'y dissout.
\end{solution}

\begin{solution}{exo:g9:acids-bases-ph:12}
Le \omterm{def:g9:acids-bases-ph:ph-paper}{papier pH} se lit à l'œil sur une échelle de couleurs, à environ une
unité près~; le \omterm{def:g9:acids-bases-ph:ph-paper}{pH-mètre} donne un nombre à une ou deux décimales. Le
\omterm{def:g9:acids-bases-ph:ph-paper}{pH-mètre} est plus précis~; les deux résultats concordent dans la
limite de précision du papier.
\end{solution}

\begin{solution}{pb:g9:acids-bases-ph:1}
\textbf{1.} Acide~: elle contient surtout des \omterm{def:g9:ions:ion}{ions} \ce{H+(aq)} et
\ce{Cl-(aq)}.

\textbf{2.} GHS05, corrosif~: un acide attaque la peau, les yeux et les
métaux.

\textbf{3.} Il donne un nombre précis au lieu d'une couleur à comparer à
l'œil.

\textbf{4.} Dix fois plus~: une unité de \omterm{def:g9:acids-bases-ph:ph}{pH} correspond à une dilution au
dixième.

\textbf{5.} \qty{10}{L}, de \omterm{def:g9:acids-bases-ph:ph}{pH} 3.

\textbf{6.} Trois~: de \omterm{def:g9:acids-bases-ph:ph}{pH} 2 à 3, de 3 à 4, de 4 à 5.

\textbf{7.} $1 \times 10 \times 10 \times 10 = \qty{1000}{L}$.

\textbf{8.} Non~: diluer ne fait que rapprocher le \omterm{def:g9:acids-bases-ph:ph}{pH} de 7.

\textbf{9.} \ce{H+ + OH- -> H2O}.

\textbf{10.} L'acide est détruit, transformé en eau, au lieu d'être
réparti dans un énorme volume~; il suffit d'un peu de base, et non de
mille litres d'eau.

\textbf{11.} Du bicarbonate de soude dissous dans l'eau (ou de l'eau
savonneuse).

\textbf{12.} $1000 - 1 = \qty{999}{L}$ d'eau.
\end{solution}
