\section*{Bab \ref{ch:g9:acids-bases-ph} --- Asam, Basa, dan pH}

\begin{solution}{exo:g9:acids-bases-ph:1}
\omterm{def:g9:acids-bases-ph:ph}{pH} 3: asam. \omterm{def:g9:acids-bases-ph:ph}{pH} 7: netral. \omterm{def:g9:acids-bases-ph:ph}{pH} 11: basa.
\end{solution}

\begin{solution}{exo:g9:acids-bases-ph:2}
Asam: \omterm{def:g9:ions:ion}{ion} \ce{H+}. Basa: \omterm{def:g9:ions:ion}{ion} \ce{OH-}.
\end{solution}

\begin{solution}{exo:g9:acids-bases-ph:3}
Air jeruk nipis (2), kopi (5), air murni (7), air laut (8.1), amonia
rumah tangga (12).
\end{solution}

\begin{solution}{exo:g9:acids-bases-ph:4}
Secarik \omterm{def:g9:acids-bases-ph:ph-paper}{kertas pH} di atas cawan yang bersih dan kering disentuh dengan
setetes larutan yang dibawa oleh batang kaca yang bersih; warnanya
segera dibandingkan dengan bagan.
\end{solution}

\begin{solution}{exo:g9:acids-bases-ph:5}
Sekitar 5 setelah pengenceran sepuluh kali lipat; sekitar 6 setelah
pengenceran seratus kali lipat.
\end{solution}

\begin{solution}{exo:g9:acids-bases-ph:6}
Tidak. Pengenceran mendekatkan \omterm{def:g9:acids-bases-ph:ph}{pH} ke 7, tetapi tidak pernah
melewatinya: air sendiri netral, dan menambahkannya tidak dapat membuat
\omterm{def:g9:ions:ion}{ion} \ce{OH-} lebih banyak daripada \omterm{def:g9:ions:ion}{ion} \ce{H+}.
\end{solution}

\begin{solution}{exo:g9:acids-bases-ph:7}
\omterm{def:g2:mixing-and-dissolving:mix}{Pencampuran} itu melepaskan panas. Air yang dituang ke asam pekat dapat
langsung mendidih dan memercikkan asam keluar; asam yang dituang
perlahan ke dalam air langsung tersebar dalam volume yang besar.
\end{solution}

\begin{solution}{exo:g9:acids-bases-ph:8}
Korosif: melukai kulit dan mata, merusak logam. Pakai kacamata pelindung
dan sarung tangan; bilas percikan apa pun dengan banyak air.
\end{solution}

\begin{solution}{exo:g9:acids-bases-ph:9}
Tiga kali pengenceran (1 ke 2, 2 ke 3, 3 ke 4): $10 \times 10 \times 10 =
1000$ kali.
\end{solution}

\begin{solution}{exo:g9:acids-bases-ph:10}
Susu magnesia bersifat basa (\omterm{def:g9:acids-bases-ph:ph}{pH} sekitar 10.5): susu magnesia bereaksi
dengan sebagian asam lambung yang berlebih.
\end{solution}

\begin{solution}{exo:g9:acids-bases-ph:11}
Tidak: dengan \omterm{def:g9:acids-bases-ph:ph}{pH} 8.1 samudra masih sedikit basa. \omterm{def:g9:acids-bases-ph:ph}{pH-nya} bergerak ke arah
yang lebih rendah, kurang basa, karena karbon dioksida dari \omterm{def:g4:air-a-mixture-of-gases:air}{udara} \omterm{def:g2:mixing-and-dissolving:dissolve}{larut}
di dalamnya.
\end{solution}

\begin{solution}{exo:g9:acids-bases-ph:12}
\omterm{def:g9:acids-bases-ph:ph-paper}{Kertas pH} dibaca dengan mata terhadap bagan warna, sampai sekitar satu
satuan; \omterm{def:g9:acids-bases-ph:ph-paper}{pH meter} memberikan angka sampai satu atau dua angka di belakang
koma. \omterm{def:g9:acids-bases-ph:ph-paper}{pH meter} lebih teliti; kedua hasil itu sesuai dalam batas
ketelitian kertas.
\end{solution}

\begin{solution}{pb:g9:acids-bases-ph:1}
\textbf{1.} Asam: larutan itu sebagian besar mengandung \omterm{def:g9:ions:ion}{ion} \ce{H+(aq)}
dan \ce{Cl-(aq)}.

\textbf{2.} GHS05, korosif: asam merusak kulit, mata, dan logam.

\textbf{3.} \omterm{def:g9:acids-bases-ph:ph-paper}{pH meter} memberikan angka yang teliti alih-alih warna yang
harus dibandingkan dengan mata.

\textbf{4.} Sepuluh kali lebih banyak: satu satuan \omterm{def:g9:acids-bases-ph:ph}{pH} sesuai dengan
pengenceran sepuluh kali lipat.

\textbf{5.} \qty{10}{L}, \omterm{def:g9:acids-bases-ph:ph}{ber-pH} 3.

\textbf{6.} Tiga: \omterm{def:g9:acids-bases-ph:ph}{pH} 2 ke 3, 3 ke 4, 4 ke 5.

\textbf{7.} $1 \times 10 \times 10 \times 10 = \qty{1000}{L}$.

\textbf{8.} Tidak: pengenceran hanya mendekatkan \omterm{def:g9:acids-bases-ph:ph}{pH} ke 7.

\textbf{9.} \ce{H+ + OH- -> H2O}.

\textbf{10.} Asamnya dihancurkan, diubah menjadi air, alih-alih disebar
ke dalam volume yang sangat besar; hanya diperlukan sedikit basa dan
tidak perlu seribu liter air.

\textbf{11.} Soda kue yang dilarutkan dalam air (atau air sabun).

\textbf{12.} $1000 - 1 = \qty{999}{L}$ air.
\end{solution}
