\section*{Chapter \ref{ch:g2:mixing-and-dissolving} --- Mixing and Dissolving}

\begin{solution}{exo:g2:mixing-and-dissolving:1}
Sugar and salt \omterm{def:g2:mixing-and-dissolving:dissolve}{dissolve}. Sand, chalk and pebbles do not.
\end{solution}

\begin{solution}{exo:g2:mixing-and-dissolving:2}
A \omterm{def:g2:mixing-and-dissolving:dissolve}{solution} is the clear liquid obtained when a solid \omterm{def:g2:mixing-and-dissolving:dissolve}{dissolves} in water.
Examples: sugar water, salty water (sweet tea is one too).
\end{solution}

\begin{solution}{exo:g2:mixing-and-dissolving:3}
Yes: oats, raisins, nuts and seeds have been put together. Every part
can still be seen and picked out.
\end{solution}

\begin{solution}{exo:g2:mixing-and-dissolving:4}
No. A dissolved solid leaves the water clear with nothing at the bottom;
here the water is cloudy and a layer has settled.
\end{solution}

\begin{solution}{exo:g2:mixing-and-dissolving:5}
It is clear (we can see through it) but not colourless (it is pink). It
is a \omterm{def:g2:mixing-and-dissolving:dissolve}{solution}: a \omterm{def:g2:mixing-and-dissolving:dissolve}{solution} may have a colour.
\end{solution}

\begin{solution}{exo:g2:mixing-and-dissolving:6}
The tea tastes sweet: the sugar is still in it, spread through the water
in pieces too small to see.
\end{solution}

\begin{solution}{exo:g2:mixing-and-dissolving:7}
The sun dries the water away; the salt that was dissolved in the sea
water is left behind.
\end{solution}

\begin{solution}{exo:g2:mixing-and-dissolving:8}
``Dissolved.'' Melting is what an ice cube does by itself, without
water; the sugar needs the water, and it is still in the tea.
\end{solution}

\begin{solution}{exo:g2:mixing-and-dissolving:9}
The beaker with the sand is cloudy and has a layer at the bottom; the
sugar water is clear with nothing at the bottom.
\end{solution}

\begin{solution}{exo:g2:mixing-and-dissolving:10}
Tap water is not only water: something is dissolved in it. When the
water dries away, that solid is left behind as the white ring.
\end{solution}

\begin{solution}{exo:g2:mixing-and-dissolving:11}
She lets a few drops of each dry on a clean dark dish. The pure water
leaves nothing; the salty water leaves white salt.
\end{solution}
