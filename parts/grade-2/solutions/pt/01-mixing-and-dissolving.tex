\section*{Capítulo \ref{ch:g2:mixing-and-dissolving} --- Misturar e dissolver}

\begin{solution}{exo:g2:mixing-and-dissolving:1}
O açúcar e o sal se dissolvem. A areia, o giz e as pedrinhas não.
\end{solution}

\begin{solution}{exo:g2:mixing-and-dissolving:2}
Uma \omterm{def:g2:mixing-and-dissolving:dissolve}{solução} é o líquido límpido obtido quando um sólido \omterm{def:g2:mixing-and-dissolving:dissolve}{se dissolve} na
água. Exemplos: a água com açúcar, a água salgada (o chá adoçado também é uma).
\end{solution}

\begin{solution}{exo:g2:mixing-and-dissolving:3}
Sim: aveia, passas, castanhas e sementes foram juntadas. Ainda dá para
ver cada parte e tirá-la de novo.
\end{solution}

\begin{solution}{exo:g2:mixing-and-dissolving:4}
Não. Um sólido dissolvido deixa a água límpida, sem nada no fundo; aqui
a água está turva e uma camada se depositou.
\end{solution}

\begin{solution}{exo:g2:mixing-and-dissolving:5}
Ele é límpido (dá para ver através dele), mas não é incolor (é
cor-de-rosa). É uma \omterm{def:g2:mixing-and-dissolving:dissolve}{solução}: uma \omterm{def:g2:mixing-and-dissolving:dissolve}{solução} pode ter cor.
\end{solution}

\begin{solution}{exo:g2:mixing-and-dissolving:6}
O chá tem gosto doce: o açúcar continua nele, espalhado pela água em
pedacinhos pequenos demais para serem vistos.
\end{solution}

\begin{solution}{exo:g2:mixing-and-dissolving:7}
O sol seca a água; o sal que estava dissolvido na água do mar fica para
trás.
\end{solution}

\begin{solution}{exo:g2:mixing-and-dissolving:8}
``Dissolveu.'' Derreter é o que um cubo de gelo faz sozinho, sem água;
o açúcar precisa da água, e ele continua no chá.
\end{solution}

\begin{solution}{exo:g2:mixing-and-dissolving:9}
O béquer com a areia está turvo e tem uma camada no fundo; a água com
açúcar está límpida, sem nada no fundo.
\end{solution}

\begin{solution}{exo:g2:mixing-and-dissolving:10}
A água da torneira não é só água: tem alguma coisa dissolvida nela.
Quando a água seca, esse sólido fica para trás, formando o anel branco.
\end{solution}

\begin{solution}{exo:g2:mixing-and-dissolving:11}
Ela deixa algumas gotas de cada um secarem num pratinho escuro e limpo.
A água pura não deixa nada; a água salgada deixa um sal branco.
\end{solution}
