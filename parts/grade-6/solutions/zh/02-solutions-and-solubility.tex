\section*{第 \ref{ch:g6:solutions-and-solubility} 章——溶液与溶解度}

\begin{solution}{exo:g6:solutions-and-solubility:1}
\omterm{def:g6:solutions-and-solubility:solute}{溶质}：糖。\omterm{def:g6:solutions-and-solubility:solute}{溶剂}：水。是的，\omterm{def:g6:solutions-and-solubility:solute}{溶剂}是水：它是一种
\omterm{def:g6:solutions-and-solubility:solute}{水溶液}。
\end{solution}

\begin{solution}{exo:g6:solutions-and-solubility:2}
一种不能再\omterm{def:g2:mixing-and-dissolving:dissolve}{溶解}\omterm{def:g6:solutions-and-solubility:solute}{溶质}的溶液。加入其中的\omterm{def:g6:solutions-and-solubility:solute}{溶质}
仍未\omterm{def:g2:mixing-and-dissolving:dissolve}{溶解}，留在底部。
\end{solution}

\begin{solution}{exo:g6:solutions-and-solubility:3}
$250 + 20 = \qty{270}{g}$.
\end{solution}

\begin{solution}{exo:g6:solutions-and-solubility:4}
乙醇：\omterm{def:g6:solutions-and-solubility:miscible}{互溶的}。食用油：\omterm{def:g6:solutions-and-solubility:miscible}{不互溶的}。醋：\omterm{def:g6:solutions-and-solubility:miscible}{互溶的}（它
主要是水）。
\end{solution}

其蒸气可能导致嗜睡或头晕。
\end{solution}

\begin{solution}{exo:g6:solutions-and-solubility:6}
$360 \times \frac{500}{1000} = \qty{180}{g}$.
\end{solution}

\begin{solution}{exo:g6:solutions-and-solubility:7}
\qty{200}{g} 水最多能\omterm{def:g2:mixing-and-dissolving:dissolve}{溶解} $360 \times \frac{200}{1000} =

\end{solution}

\begin{solution}{exo:g6:solutions-and-solubility:8}

\end{solution}

\begin{solution}{exo:g6:solutions-and-solubility:9}
在第一个烧杯中，盐已全部\omterm{def:g2:mixing-and-dissolving:dissolve}{溶解}，溶液
尚未饱和。
\end{solution}

\begin{solution}{exo:g6:solutions-and-solubility:10}
二氧化碳是在密闭瓶中，于较高压强下\omterm{def:g2:mixing-and-dissolving:dissolve}{溶解}的。
打开瓶盖会降低压强：水能容纳的气体变少，
多出的气体以气泡的形式逸出。
\end{solution}

\begin{solution}{exo:g6:solutions-and-solubility:11}
$2000 \div 0.04 = 50\,000$ 倍。
\end{solution}

\begin{solution}{exo:g6:solutions-and-solubility:12}
气体在温水中\omterm{def:g2:mixing-and-dissolving:dissolve}{溶解}得较少：温暖的池塘所含的\omterm{def:g2:mixing-and-dissolving:dissolve}{溶解}氧较少，
鱼就会缺氧。
\end{solution}

\begin{solution}{pb:g6:solutions-and-solubility:1}
\textbf{1.} \omterm{def:g6:solutions-and-solubility:solute}{溶质}：盐。\omterm{def:g6:solutions-and-solubility:solute}{溶剂}：水。

\textbf{2.} 水已\omterm{def:g2:mixing-and-dissolving:dissolve}{溶解}了最大量的盐：
不能再\omterm{def:g2:mixing-and-dissolving:dissolve}{溶解}更多的盐。

\textbf{3.} 底部未\omterm{def:g2:mixing-and-dissolving:dissolve}{溶解}的盐说明盐水
已饱和：它所\omterm{def:g2:mixing-and-dissolving:dissolve}{溶解}的盐已达到最大量。

\textbf{4.} $360 \times 20 = \qty{7200}{g} = \qty{7.2}{kg}$.

\textbf{5.} $20 + 7.2 = \qty{27.2}{kg}$.

\textbf{6.} $\frac{7.2}{27.2} \approx 0.26$：约 \qty{26}{\percent}。

\textbf{7.} 否：水原本能\omterm{def:g2:mixing-and-dissolving:dissolve}{溶解} \qty{7.2}{kg}。
仍可再\omterm{def:g2:mixing-and-dissolving:dissolve}{溶解} $7.2 - 5 = \qty{2.2}{kg}$。

\textbf{8.} $20 - 5 = \qty{15}{kg}$ 的水（\qty{15}{L}）。

\textbf{9.} $360 \times 15 = \qty{5400}{g} = \qty{5.4}{kg}$.

\textbf{10.} 它以固态晶体的形式从溶液中析出：它
结晶后下沉到底部。

\textbf{11.} 新的白色晶体出现，并在
水槽底部堆积。

\textbf{12.} $7.2 - 5.4 = \qty{1.8}{kg}$ 的盐已结晶。
\end{solution}
