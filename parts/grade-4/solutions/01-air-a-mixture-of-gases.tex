\section*{Chapter \ref{ch:g4:air-a-mixture-of-gases} --- Air, a Mixture of Gases}

\begin{solution}{exo:g4:air-a-mixture-of-gases:1}
A mixture. Its two main gases are nitrogen and oxygen.
\end{solution}

\begin{solution}{exo:g4:air-a-mixture-of-gases:2}
About four fifths of the \omterm{def:g4:air-a-mixture-of-gases:air}{air} is nitrogen, about one fifth is oxygen.
\end{solution}

\begin{solution}{exo:g4:air-a-mixture-of-gases:3}
About 78 litres of nitrogen and about 21 litres of oxygen.
\end{solution}

\begin{solution}{exo:g4:air-a-mixture-of-gases:4}
The bottle is not empty: it is full of \omterm{def:g4:air-a-mixture-of-gases:air}{air}, and the trapped \omterm{def:g4:air-a-mixture-of-gases:air}{air} keeps
the water out.
\end{solution}

\begin{solution}{exo:g4:air-a-mixture-of-gases:5}
There is less oxygen (16 parts in 100 instead of 21) and more carbon
dioxide (4 parts in 100 instead of almost none).
\end{solution}

\begin{solution}{exo:g4:air-a-mixture-of-gases:6}
Oxygen in both cases: a flame and our body both use the oxygen of the
\omterm{def:g4:air-a-mixture-of-gases:air}{air}.
\end{solution}

\begin{solution}{exo:g4:air-a-mixture-of-gases:7}
78 squares are nitrogen; $100 - 78 = 22$ squares are not.
\end{solution}

\begin{solution}{exo:g4:air-a-mixture-of-gases:8}
The flame uses the oxygen of the \omterm{def:g4:air-a-mixture-of-gases:air}{air} in the jar. When too little oxygen
is left around the flame, it goes out.
\end{solution}

\begin{solution}{exo:g4:air-a-mixture-of-gases:9}
One fifth of 500 litres: $500 \div 5 = 100$ litres of oxygen.
\end{solution}

\begin{solution}{exo:g4:air-a-mixture-of-gases:10}
The people in the room breathe in oxygen and breathe out carbon
dioxide. Opening the windows brings back fresh \omterm{def:g4:air-a-mixture-of-gases:air}{air}, with more oxygen and
less carbon dioxide.
\end{solution}

\begin{solution}{exo:g4:air-a-mixture-of-gases:11}
Three times as much \omterm{def:g4:air-a-mixture-of-gases:air}{air} holds three times as much oxygen: about
$3 \times 10 = 30$ seconds.
\end{solution}
