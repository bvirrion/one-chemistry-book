\section*{Chapter \ref{ch:g4:what-a-fire-needs} --- Burning: What a Fire Needs}

\begin{solution}{exo:g4:what-a-fire-needs:1}
\omterm{def:g4:what-a-fire-needs:fuel}{Fuel}, oxygen (from the \omterm{def:g4:air-a-mixture-of-gases:air}{air}) and heat (to start the fire).
\end{solution}

\begin{solution}{exo:g4:what-a-fire-needs:2}
Wood, candle wax, petrol and paper are \omterm{def:g4:what-a-fire-needs:fuel}{fuels}. Sand, glass and water do
not burn.
\end{solution}

\begin{solution}{exo:g4:what-a-fire-needs:3}
The oxygen: the blanket stops the \omterm{def:g4:air-a-mixture-of-gases:air}{air} from reaching the fire.
\end{solution}

\begin{solution}{exo:g4:what-a-fire-needs:4}
The heat: the water cools the \omterm{def:g4:what-a-fire-needs:burning}{burning} wood so much that it can no
longer burn.
\end{solution}

\begin{solution}{exo:g4:what-a-fire-needs:5}
Any three of: ash, smoke, carbon dioxide, water vapour.
\end{solution}

\begin{solution}{exo:g4:what-a-fire-needs:6}
Blowing brings more \omterm{def:g4:air-a-mixture-of-gases:air}{air}, so more oxygen, to the hot \omterm{def:g4:what-a-fire-needs:fuel}{fuel}, and it burns
again.
\end{solution}

\begin{solution}{exo:g4:what-a-fire-needs:7}
Water: the \omterm{def:g4:what-a-fire-needs:burning}{burning} wax makes water vapour, which turns into droplets on
the cold glass.
\end{solution}

\begin{solution}{exo:g4:what-a-fire-needs:8}
Get out, keeping low under the smoke and closing the doors behind you,
then call the fire brigade from outside.
\end{solution}

\begin{solution}{exo:g4:what-a-fire-needs:9}
The \omterm{def:g4:what-a-fire-needs:fuel}{fuel} side: with the gas turned off, the fire has nothing left to
burn.
\end{solution}

\begin{solution}{exo:g4:what-a-fire-needs:10}
The \omterm{def:g4:what-a-fire-needs:fuel}{fuel}: with no trees in the strip, the fire runs out of \omterm{def:g4:what-a-fire-needs:fuel}{fuel} when it
reaches it.
\end{solution}

\begin{solution}{exo:g4:what-a-fire-needs:11}
No. Most of the wood, together with oxygen of the \omterm{def:g4:air-a-mixture-of-gases:air}{air}, has become new
substances that are gases --- carbon dioxide, water vapour and smoke ---
and they have gone into the \omterm{def:g4:air-a-mixture-of-gases:air}{air}. Only the ash stays behind.
\end{solution}
