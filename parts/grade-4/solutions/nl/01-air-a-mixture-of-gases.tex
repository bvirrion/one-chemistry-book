\section*{Hoofdstuk \ref{ch:g4:air-a-mixture-of-gases} --- Lucht, een mengsel van gassen}

\begin{solution}{exo:g4:air-a-mixture-of-gases:1}
Een mengsel. De twee belangrijkste gassen zijn stikstof en zuurstof.
\end{solution}

\begin{solution}{exo:g4:air-a-mixture-of-gases:2}
Ongeveer vier vijfde van de \omterm{def:g4:air-a-mixture-of-gases:air}{lucht} is stikstof, ongeveer een vijfde is
zuurstof.
\end{solution}

\begin{solution}{exo:g4:air-a-mixture-of-gases:3}
Ongeveer 78 liter stikstof en ongeveer 21 liter zuurstof.
\end{solution}

\begin{solution}{exo:g4:air-a-mixture-of-gases:4}
De fles is niet leeg: hij zit vol \omterm{def:g4:air-a-mixture-of-gases:air}{lucht}, en de opgesloten \omterm{def:g4:air-a-mixture-of-gases:air}{lucht} houdt
het water tegen.
\end{solution}

\begin{solution}{exo:g4:air-a-mixture-of-gases:5}
Er zit minder zuurstof in (16 delen op 100 in plaats van 21) en meer
koolstofdioxide (4 delen op 100 in plaats van bijna niets).
\end{solution}

\begin{solution}{exo:g4:air-a-mixture-of-gases:6}
In beide gevallen zuurstof: een vlam en ons lichaam gebruiken allebei de
zuurstof van de \omterm{def:g4:air-a-mixture-of-gases:air}{lucht}.
\end{solution}

\begin{solution}{exo:g4:air-a-mixture-of-gases:7}
78 vakjes zijn stikstof; $100 - 78 = 22$ vakjes niet.
\end{solution}

\begin{solution}{exo:g4:air-a-mixture-of-gases:8}
De vlam verbruikt de zuurstof van de \omterm{def:g4:air-a-mixture-of-gases:air}{lucht} in de stolp. Als er rond de
vlam te weinig zuurstof over is, gaat hij uit.
\end{solution}

\begin{solution}{exo:g4:air-a-mixture-of-gases:9}
Een vijfde van 500 liter: $500 \div 5 = 100$ liter zuurstof.
\end{solution}

\begin{solution}{exo:g4:air-a-mixture-of-gases:10}
De mensen in het lokaal ademen zuurstof in en koolstofdioxide uit. Als
je de ramen openzet, komt er weer frisse \omterm{def:g4:air-a-mixture-of-gases:air}{lucht} binnen, met meer zuurstof
en minder koolstofdioxide.
\end{solution}

\begin{solution}{exo:g4:air-a-mixture-of-gases:11}
Drie keer zoveel \omterm{def:g4:air-a-mixture-of-gases:air}{lucht} bevat drie keer zoveel zuurstof: ongeveer
$3 \times 10 = 30$ seconden.
\end{solution}
