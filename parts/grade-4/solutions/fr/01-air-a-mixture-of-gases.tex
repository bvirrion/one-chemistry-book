\section*{Chapitre \ref{ch:g4:air-a-mixture-of-gases} --- L'air, un mélange de gaz}

\begin{solution}{exo:g4:air-a-mixture-of-gases:1}
Un \omterm{def:g2:mixing-and-dissolving:mix}{mélange}. Ses deux principaux gaz sont l'azote et l'oxygène.
\end{solution}

\begin{solution}{exo:g4:air-a-mixture-of-gases:2}
Environ les quatre cinquièmes de l'\omterm{def:g4:air-a-mixture-of-gases:air}{air} sont de l'azote, environ un
cinquième de l'oxygène.
\end{solution}

\begin{solution}{exo:g4:air-a-mixture-of-gases:3}
Environ 78 litres d'azote et environ 21 litres d'oxygène.
\end{solution}

\begin{solution}{exo:g4:air-a-mixture-of-gases:4}
La bouteille n'est pas vide~: elle est pleine d'\omterm{def:g4:air-a-mixture-of-gases:air}{air}, et l'\omterm{def:g4:air-a-mixture-of-gases:air}{air} emprisonné
empêche l'eau d'entrer.
\end{solution}

\begin{solution}{exo:g4:air-a-mixture-of-gases:5}
Il y a moins d'oxygène (16 parts sur 100 au lieu de 21) et plus de
dioxyde de carbone (4 parts sur 100 au lieu de presque rien).
\end{solution}

\begin{solution}{exo:g4:air-a-mixture-of-gases:6}
L'oxygène dans les deux cas~: une flamme et notre corps utilisent tous
deux l'oxygène de l'\omterm{def:g4:air-a-mixture-of-gases:air}{air}.
\end{solution}

\begin{solution}{exo:g4:air-a-mixture-of-gases:7}
78 cases sont de l'azote~; $100 - 78 = 22$ cases n'en sont pas.
\end{solution}

\begin{solution}{exo:g4:air-a-mixture-of-gases:8}
La flamme consomme l'oxygène de l'\omterm{def:g4:air-a-mixture-of-gases:air}{air} du bocal. Quand il reste trop peu
d'oxygène autour de la flamme, elle s'éteint.
\end{solution}

\begin{solution}{exo:g4:air-a-mixture-of-gases:9}
Un cinquième de 500 litres~: $500 \div 5 = 100$ litres d'oxygène.
\end{solution}

\begin{solution}{exo:g4:air-a-mixture-of-gases:10}
Les personnes présentes inspirent de l'oxygène et expirent du dioxyde de
carbone. Ouvrir les fenêtres fait revenir de l'\omterm{def:g4:air-a-mixture-of-gases:air}{air} frais, avec plus
d'oxygène et moins de dioxyde de carbone.
\end{solution}

\begin{solution}{exo:g4:air-a-mixture-of-gases:11}
Trois fois plus d'\omterm{def:g4:air-a-mixture-of-gases:air}{air} contient trois fois plus d'oxygène~: environ
$3 \times 10 = 30$ secondes.
\end{solution}
