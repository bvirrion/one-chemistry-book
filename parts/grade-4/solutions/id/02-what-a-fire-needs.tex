\section*{Bab \ref{ch:g4:what-a-fire-needs} --- Pembakaran: Apa yang Diperlukan Api}

\begin{solution}{exo:g4:what-a-fire-needs:1}
\omterm{def:g4:what-a-fire-needs:fuel}{Bahan bakar}, oksigen (dari \omterm{def:g4:air-a-mixture-of-gases:air}{udara}), dan panas (untuk menyalakan api).
\end{solution}

\begin{solution}{exo:g4:what-a-fire-needs:2}
Kayu, lilin, bensin, dan kertas adalah \omterm{def:g4:what-a-fire-needs:fuel}{bahan bakar}. Pasir, kaca, dan
air tidak dapat \omterm{def:g4:what-a-fire-needs:burning}{terbakar}.
\end{solution}

\begin{solution}{exo:g4:what-a-fire-needs:3}
Oksigen: selimut itu menghalangi \omterm{def:g4:air-a-mixture-of-gases:air}{udara} mencapai api.
\end{solution}

\begin{solution}{exo:g4:what-a-fire-needs:4}
Panas: air mendinginkan kayu yang \omterm{def:g4:what-a-fire-needs:burning}{terbakar} begitu rupa sehingga kayu
itu tidak dapat \omterm{def:g4:what-a-fire-needs:burning}{terbakar} lagi.
\end{solution}

\begin{solution}{exo:g4:what-a-fire-needs:5}
Tiga di antara: abu, asap, karbon dioksida, uap air.
\end{solution}

\begin{solution}{exo:g4:what-a-fire-needs:6}
Tiupan membawa lebih banyak \omterm{def:g4:air-a-mixture-of-gases:air}{udara}, jadi lebih banyak oksigen, ke \omterm{def:g4:what-a-fire-needs:fuel}{bahan
bakar} yang panas, dan \omterm{def:g4:what-a-fire-needs:fuel}{bahan bakar} itu \omterm{def:g4:what-a-fire-needs:burning}{terbakar} lagi.
\end{solution}

\begin{solution}{exo:g4:what-a-fire-needs:7}
Air: lilin yang \omterm{def:g4:what-a-fire-needs:burning}{terbakar} menghasilkan uap air, yang berubah menjadi
titik-titik air pada kaca yang dingin.
\end{solution}

\begin{solution}{exo:g4:what-a-fire-needs:8}
Keluar, sambil tetap merunduk di bawah asap dan menutup pintu di
belakangmu, lalu menelepon pemadam kebakaran dari luar.
\end{solution}

\begin{solution}{exo:g4:what-a-fire-needs:9}
Sisi \omterm{def:g4:what-a-fire-needs:fuel}{bahan bakar}: dengan gas dimatikan, api tidak punya apa-apa lagi
untuk dibakar.
\end{solution}

\begin{solution}{exo:g4:what-a-fire-needs:10}
\omterm{def:g4:what-a-fire-needs:fuel}{Bahan bakar}: karena tidak ada pohon di jalur itu, api kehabisan \omterm{def:g4:what-a-fire-needs:fuel}{bahan
bakar} ketika sampai di sana.
\end{solution}

\begin{solution}{exo:g4:what-a-fire-needs:11}
Tidak. Sebagian besar kayu itu, bersama oksigen dari \omterm{def:g4:air-a-mixture-of-gases:air}{udara}, telah
menjadi zat-zat baru yang berupa gas --- karbon dioksida, uap air, dan
asap --- dan semuanya telah pergi ke \omterm{def:g4:air-a-mixture-of-gases:air}{udara}. Hanya abunya yang
tertinggal.
\end{solution}
