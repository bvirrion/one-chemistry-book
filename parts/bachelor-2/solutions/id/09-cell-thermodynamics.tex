\begin{solution}{exo:b2:cell-thermodynamics:2}

\begin{solution}{exo:b2:cell-thermodynamics:3}
$\Delta_r G = -2 \times 96\,485 \times 1.10 = \qty{-212}{kJ/mol}$. Untuk
\qty{10.0}{g}, $10.0/65.38 = \qty{0.153}{mol}$: paling banyak \qty{32.5}{kJ}.
\end{solution}

\begin{solution}{exo:b2:cell-thermodynamics:4}
$U = (0.0592/2)\log(0.10/1.0 \times 10^{-3}) = \qty{0.059}{V}$; kutub
positif (katode) adalah elektrode di dalam larutan \qty{0.10}{mol/L}.
\end{solution}

\begin{solution}{exo:b2:cell-thermodynamics:5}
$\Delta_r G^\circ = -2 \times 96\,485 \times 1.015 = \qty{-195.9}{kJ/mol}$;
$\Delta_r S^\circ = 2 \times 96\,485 \times (-4.0 \times 10^{-4}) =
\qty{-77.2}{J/(K.mol)}$; $\Delta_r H^\circ = -195.9 + 298 \times (-0.0772) =
\qty{-218.9}{kJ/mol}$.
\end{solution}

\begin{solution}{exo:b2:cell-thermodynamics:6}
(a) Kerja yang diberikan \qty{195.9}{kJ}; kalor $T\Delta_r S = \qty{-23.0}{kJ}$:
\qty{23.0}{kJ} dilepaskan. (b) Kerja $2 \times 96\,485 \times 0.90 = \qty{173.7}{kJ}$; kalor yang dilepaskan $-\Delta_r H - 173.7 = 218.9 - 173.7 = \qty{45.2}{kJ}$. Tambahan \qty{22.2}{kJ} itu adalah kerja yang hilang
karena arus.
\end{solution}

\begin{solution}{exo:b2:cell-thermodynamics:7}
\ce{Fe^3+ + 3e- -> Fe}: $\Delta_r G^\circ = +\qty{4.7}{kJ/mol}$, $E^\circ =
-4700/(3 \times 96\,485) = \qty{-0.016}{V}$. Dengan $E_1^\circ = 0.769$ dan
$E_2^\circ = \qty{-0.409}{V}$: $(0.769 - 0.818)/3 = \qty{-0.016}{V}$.
\end{solution}

\begin{solution}{exo:b2:cell-thermodynamics:8}

$E = 0.222 - 0.0592\log 0.10 =
\qty{0.281}{V}$.
\end{solution}

\begin{solution}{exo:b2:cell-thermodynamics:9}
$\Delta_r G^\circ = 2(-318.30) = \qty{-636.6}{kJ}$ untuk $n = 4$: $E^\circ =
636\,600/(4 \times 96\,485) = \qty{1.65}{V}$. Per kilogram seng,
$1000/65.38 =
\qty{15.3}{mol}$, $15.3 \times 318.3 = \qty{4.87}{MJ}$, yaitu
\qty{1.35}{kW.h}.
\end{solution}

\begin{solution}{exo:b2:cell-thermodynamics:10}
Karbon berubah dari $-2$ menjadi $+4$: $n = 6$. $\Delta_r H^\circ = -393.51 + 2(-285.83) +
239 = \qty{-726.2}{kJ/mol}$; $\Delta_r S^\circ = 213.8 + 2(69.95) - 127.2 - 1.5
\times 205.15 = \qty{-81.2}{J/(K.mol)}$; $\Delta_r G^\circ = -726.2 + 298.15 \times
0.0812 = \qty{-702.0}{kJ/mol}$. $E^\circ = 702\,000/(6 \times 96\,485) =
\qty{1.21}{V}$; efisiensi maksimum $702.0/726.2 = 0.967$. (Sel metanol
nyata bekerja jauh di bawah tegangan ini.)
\end{solution}

\begin{solution}{exo:b2:cell-thermodynamics:11}
$\Delta_r G = -2 \times 96\,485 \times 0.500 = \qty{-96.5}{kJ}$;
$\Delta_r S = +\qty{57.9}{J/K}$;
$\Delta_r H = -96.5 + 298 \times 0.0579 = \qty{-79.2}{kJ}$. Kerjanya,
\qty{96.5}{kJ}, melampaui $-\Delta_r H = \qty{79.2}{kJ}$: sel menyerap
$T\Delta_r S = \qty{17.3}{kJ}$ kalor dari lingkungannya dan juga
mengubahnya menjadi kerja. Hukum kedua tidak dilanggar, karena \omterm{def:b2:reaction-free-energy:reaction-entropy}{entropi
reaksi} bertambah sebesar itu pula.
\end{solution}

\begin{solution}{exo:b2:cell-thermodynamics:12}
Setiap elektrode hidrogen mempunyai $E = -0.0592\,\mathrm{pH}$:
$U = 0.0592\,(\mathrm{pH}_s -
7.00) = 0.177$, $\mathrm{pH}_s = 10.0$. Kedua
elektrode berasal dari pasangan yang sama: potensial standarnya saling
meniadakan, seperti pada setiap \omterm{def:b2:cell-thermodynamics:concentration-cell}{sel konsentrasi}.
\end{solution}

\begin{solution}{pb:b2:cell-thermodynamics:1}
\textbf{1.} Anode: \ce{H2 -> 2H+ + 2e-}; katode: \ce{1/2O2 + 2H+ + 2e- -> H2O};
sel: \ce{H2 + 1/2O2 -> H2O}.
\textbf{2.} Dua.
\textbf{3.} $E^\circ = 237\,140/(2 \times 96\,485) = \qty{1.229}{V}$.
\textbf{4.} $228\,580/192\,970 = \qty{1.185}{V}$.
\textbf{5.} $\Delta_r G^\circ$ berbeda sebesar
$\Delta_{\text{vap}}G^\circ(\ce{H2O}) = -228.58 + 237.14 =
\qty{8.56}{kJ/mol}$
pada \qty{298}{K}: di sana uap kurang stabil daripada cairan, dan
tegangannya lebih rendah sebesar $8560/192\,970 = \qty{0.044}{V}$.
\textbf{6.} $\Delta_r H^\circ = \qty{-285.83}{kJ/mol}$; $237.14/285.83 = 0.830$.
\textbf{7.} $\Delta_r S^\circ = 69.95 - 130.68 - 102.58 = \qty{-163.3}{J/(K.mol)}$.
\textbf{8.} $dE^\circ/dT = -163.3/192\,970 = \qty{-8.46e-4}{V/K}$.
\textbf{9.} $1.229 - 8.46 \times 10^{-4} \times 55 = \qty{1.182}{V}$.
\textbf{10.} Perkiraan yang sama pada \qty{400}{K} memberikan
$1.229 - 8.46 \times 10^{-4}
\times 101.85 = \qty{1.143}{V}$; JANAF
memberikan $221\,010/192\,970 = \qty{1.145}{V}$: menganggap
$\Delta_r S^\circ$ tetap hanya meleset \qty{2}{mV}.
\textbf{11.} $T\Delta_r S^\circ = 298.15 \times (-163.3) = \qty{-48.7}{kJ}$
per mol: \qty{48.7}{kJ} dilepaskan.
\textbf{12.} $\Delta_r G^\circ(353) \approx -237.14 + 55 \times 0.1633 =
\qty{-228.2}{kJ}$; dengan $\Delta_r H^\circ$ yang hampir tidak berubah,
$228.2/285.8 = 0.80$.
\textbf{13.} $2 \times 96\,485 \times 0.70 = \qty{135.1}{kJ}$.
\textbf{14.} $135.1/285.83 = 0.47$; $135.1/237.14 = 0.57$.
\textbf{15.} $285.83 - 135.1 = \qty{150.7}{kJ}$.
\textbf{16.} Sebanyak \qty{48.7}{kJ} akan dilepaskan bahkan oleh sel
reversibel; \qty{102.0}{kJ} sisanya berasal dari kerugian akibat arus
(hambatan dan overpotensial elektrode).
\textbf{17.} Satu sel: $300/(2 \times 96\,485) = \qty{1.55e-3}{mol/s}$;
tumpukan: $400 \times 1.55 \times 10^{-3} = \qty{0.622}{mol/s}$,
\qty{1.25}{g/s}.
\textbf{18.} $400 \times 0.70 \times 300 = \qty{84}{kW}$.
\textbf{19.} $0.622 \times 150.7 = \qty{93.7}{kW}$: kalornya lebih banyak
daripada listriknya.
\textbf{20.} $5000/2.016 = \qty{2.48e3}{mol}$.
\textbf{21.} $2 \times 2480 \times 96\,485 = \qty{4.79e8}{C}$.
\textbf{22.} Setiap coulomb yang melewati satu sel memberikan
\qty{0.70}{J}: $4.79 \times
10^8 \times 0.70 = \qty{3.35e8}{J}$.
\textbf{23.} $3.35 \times 10^8/3.6 \times 10^6 = \qty{93}{kW.h}$.
\textbf{24.} $93/84 = \qty{1.1}{h}$ pada daya penuh.
\textbf{25.} Sebanyak \qty{5.0}{kg} hidrogen pada \qty{0.70}{V} per sel
memberikan $\boldsymbol{\approx \qty{93}{kW.h}}$ energi listrik.
\end{solution}
