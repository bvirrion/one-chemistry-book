\section*{Chapter \ref{ch:g5:raw-materials-and-recycling} --- From Raw Materials to Recycling}

\begin{solution}{exo:g5:raw-materials-and-recycling:1}
Natural: wood, wool, stone. Manufactured: glass, steel, plastic.
\end{solution}

\begin{solution}{exo:g5:raw-materials-and-recycling:2}
Paper: wood (trees). Glass: sand (with soda and limestone). Cotton:
the cotton plant. Plastic: crude oil.
\end{solution}

\begin{solution}{exo:g5:raw-materials-and-recycling:3}
A rock from which a metal is extracted. Aluminium comes from bauxite.
\end{solution}

\begin{solution}{exo:g5:raw-materials-and-recycling:4}
The jar in the glass bin, the newspaper with paper and card, the can
with metal, the bottle with plastic bottles.
\end{solution}

\begin{solution}{exo:g5:raw-materials-and-recycling:5}
New \omterm{def:g5:raw-materials-and-recycling:raw-material}{raw material}: the recycled material is used instead of sand, \omterm{def:g5:raw-materials-and-recycling:raw-material}{ore} or
oil taken from nature.
\end{solution}

\begin{solution}{exo:g5:raw-materials-and-recycling:6}
Renewable: wood (if replanted), wool, cotton. Not renewable: crude oil,
iron \omterm{def:g5:raw-materials-and-recycling:raw-material}{ore}.
\end{solution}

\begin{solution}{exo:g5:raw-materials-and-recycling:7}
Making glass from sand is a \omterm{def:g5:irreversible-changes:chemical-change}{chemical change}: something new is made, and
there is no simple way back.
\end{solution}

\begin{solution}{exo:g5:raw-materials-and-recycling:8}
For instance: reduce --- buy fruit without a plastic wrapping; reuse ---
refill a glass jar; recycle --- put a drink can in the metal bin.
\end{solution}

\begin{solution}{exo:g5:raw-materials-and-recycling:9}
Each material is recycled in its own way: pieces of another material
would spoil the melted glass.
\end{solution}

\begin{solution}{exo:g5:raw-materials-and-recycling:10}
One twentieth of 20 units: $20 \div 20 = 1$ unit of energy, about.
\end{solution}

\begin{solution}{exo:g5:raw-materials-and-recycling:11}
\omterm{def:g5:raw-materials-and-recycling:recycling}{Recycling} still needs energy, machines and transport; an object that is
never made saves its \omterm{def:g5:raw-materials-and-recycling:raw-material}{raw material} and all that energy.
\end{solution}
