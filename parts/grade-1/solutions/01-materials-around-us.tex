\section*{Chapter \ref{ch:g1:materials-around-us} --- What Things Are Made Of}

\begin{solution}{exo:g1:materials-around-us:1}
A chair is an \omterm{def:g1:materials-around-us:object}{object}. Wood is a \omterm{def:g1:materials-around-us:material}{material}: a chair can be made of it.
\end{solution}

\begin{solution}{exo:g1:materials-around-us:2}
Window pane: glass. Nail: metal. Pebble: stone. Book: paper. Sock:
fabric (wool or cotton).
\end{solution}

\begin{solution}{exo:g1:materials-around-us:3}
Wood: a pencil, a door (or a chair, a table). Plastic: a bowl, a bucket
(or a toy, a bottle).
\end{solution}

\begin{solution}{exo:g1:materials-around-us:4}
The metal fork: it is the only one not made of wood or plastic. (Two are
made of wood and one of plastic.)
\end{solution}

\begin{solution}{exo:g1:materials-around-us:5}
Glass: it is transparent.
\end{solution}

\begin{solution}{exo:g1:materials-around-us:6}
The steel nail. A magnet pulls iron and steel, not glass or wood.
\end{solution}

\begin{solution}{exo:g1:materials-around-us:7}
Two \omterm{def:g1:materials-around-us:object}{objects} float (the wooden block and the plastic cap); three sink
(the pebble, the marble and the nail).
\end{solution}

\begin{solution}{exo:g1:materials-around-us:8}
The blades are metal, the handles plastic. The scissors are made of two
\omterm{def:g1:materials-around-us:material}{materials}.
\end{solution}

\begin{solution}{exo:g1:materials-around-us:9}
In the blue hoop: it is made of metal. It is not in the orange hoop
because a magnet does not pull it: it is made of aluminium, not of iron
or steel.
\end{solution}

\begin{solution}{exo:g1:materials-around-us:10}
A window must let light in, and we must see through it. Glass is
transparent; wood is not.
\end{solution}

\begin{solution}{exo:g1:materials-around-us:11}
``Yes'': the nail, the paper clip and the key, 3 \omterm{def:g1:materials-around-us:object}{objects}. ``No'': the
pebble, the spoon, the can, the marble and the sock, 5 \omterm{def:g1:materials-around-us:object}{objects}.
$3 + 5 = 8$: every \omterm{def:g1:materials-around-us:object}{object} has its place.
\end{solution}
