\section*{Bab \ref{ch:b3:partition-functions} --- Termodinamika Statistik: Fungsi Partisi}

\begin{solution}{exo:b3:partition-functions:1}
Tuliskan setiap distribusi sebagai kuantum yang dipegang keempat osilator,
dari yang terbesar. ``4, 0, 0, 0'': $4!/(3!\,1!) = 4$; ``3, 1, 0, 0'':
$4!/(2!\,1!\,1!) = 12$; ``2, 2, 0, 0'':
$4!/(2!\,2!) = 6$; ``2, 1, 1, 0'': $4!/(1!\,2!\,1!) = 12$; ``1, 1, 1, 1'': 1. Jumlahnya 35 $= \binom73$.
Yang paling mungkin adalah ``3, 1, 0, 0'' dan ``2, 1, 1, 0'' (masing-masing 12);
yang kedua, dengan \omterm{def:b3:partition-functions:boltzmann}{populasi} 1, 2, 1 untuk 0, 1, 2 kuantum, sudah menurun
terhadap energi seperti \omterm{def:b3:partition-functions:boltzmann}{distribusi Boltzmann} untuk bilangan yang besar.
\end{solution}

\begin{solution}{exo:b3:partition-functions:7}
$\ln q = \ln V + \frac32\ln T + \text{tetapan}$, sehingga $U - U(0) = NkT^2 \times \frac{3}{2T} = \frac32NkT$;
untuk satu mol $\frac32RT = \qty{3.72}{kJ/mol}$ pada \qty{298.15}{K}, dan
$C_V = \frac32R = \qty{12.47}{J.K^{-1}.mol^{-1}}$.
\end{solution}

\begin{solution}{exo:b3:partition-functions:10}
$\frac{\dd}{\dd J}\bigl[(2J+1)\eu^{-\theta J(J+1)/T}\bigr] = \bigl[2 - (2J+1)^2\theta/T\bigr]\eu^{-\theta J(J+1)/T} = 0$
memberikan $2J + 1 = \sqrt{2T/\theta}$, yaitu $J_{\max} = \sqrt{T/2\theta} - \frac12$.
\ce{HCl}: 2.66, jadi $J = 3$; \ce{CO}: 6.86, jadi $J = 7$. Intensitas sebuah
garis \omterm{def:b3:rovibrational-spectroscopy:branches}{cabang P} atau R mengikuti \omterm{def:b3:partition-functions:boltzmann}{populasi} tingkat bawahnya, yang memuncak pada
$J_{\max}$, beberapa garis dari celah di pusat pita.
\end{solution}

\begin{solution}{exo:b3:partition-functions:11}
$\int_0^\infty f\dd J = T/\theta$, $f(0) = 1$, $f'(0) = 2 - \theta/T$. Jadi $q^{\mathrm R} \approx T/\theta + \frac12 -
\frac16 + \frac{\theta}{12T} \approx T/\theta + \frac13$. Galat relatif $T/\theta$
sekitar $\theta/3T$: di bawah 1\,\% jika $T > 33\,\theta$. Untuk \ce{H2},
$\theta = 1.4388 \times 59.32 = \qty{85.3}{K}$: pada \qty{298.15}{K} galatnya
10\,\%, tidak dapat diterima (dan pembatasan $J$ oleh spin inti juga harus
diperlakukan secara eksak, \cref{ch:b3:statistical-thermo-applied}).
\end{solution}

\begin{solution}{exo:b3:partition-functions:12}
Untuk $x = \theta_{\mathrm V}/T \to \infty$, $x/(\eu^x - 1) \to 0$ dan $\ln(1 - \eu^{-x}) \to 0$: $S \to 0$.
Untuk $x \to 0$,
$x/(\eu^x - 1) \approx 1 - x/2$ dan $-\ln(1 - \eu^{-x}) \approx -\ln x + x/2$: $S \to R(1 - \ln x) = R[1 + \ln(T/\theta_{\mathrm V})]$.
\ce{I2}, $x = 1.029$: eksak $R(0.5721 + 0.4420) = \qty{8.43}{J.K^{-1}.mol^{-1}}$;
limit $R(1 - 0.0289) =
\qty{8.07}{J.K^{-1}.mol^{-1}}$. Bahkan pada $T \approx \theta_{\mathrm V}$, limit
klasik itu hanya 4\,\% terlalu rendah.
\end{solution}
