\begin{solution}{exo:b3:electronic-spectroscopy:6}
Faktor Poisson $\eu^{-S}S^v/v!$ maksimum pada $v = \lfloor S\rfloor$. $S = 0.3$:
garis 0--0 itu sendiri. $S = 1.5$: $v' = 1$, 1.5 kali garis 0--0. $S = 5$:
$v' = 4$ dan 5 sama kuat, 26 kali garis 0--0.
\end{solution}

\begin{solution}{exo:b3:electronic-spectroscopy:9}
$\tau_0/\tau = 1.429$ dan 1.860: linear terhadap $[Q]$, jadi umur itu sendiri

\begin{solution}{exo:b3:electronic-spectroscopy:10}
\omterm{def:b3:electronic-spectroscopy:quencher}{Pemadaman statis}: molekul-molekul yang masih memancar hidup selama
sebelumnya; separuhnya terikat, dalam keadaan dasar, dalam kompleks yang tak
berfluoresensi. Dengan tetapan asosiasi $K_a$, fraksi bebasnya adalah
$1/(1 + K_a[Q])$ dan $I_0/I = 1 +
K_a[Q]$, bentuk yang sama dengan Stern--Volmer tetapi dengan $\tau$ yang tidak
berubah.
\end{solution}

\begin{solution}{exo:b3:electronic-spectroscopy:11}

terukur diperpendek oleh \omterm{def:b3:electronic-spectroscopy:radiationless}{lintas antarsistem} yang bersaing; $\tau_r$ adalah umur

\end{solution}

\begin{solution}{exo:b3:electronic-spectroscopy:12}
$\frac12\langle\alpha\beta - \beta\alpha|\alpha\beta + \beta\alpha\rangle = \frac12(1 + 0 - 0 - 1) = 0$,
dengan memakai $\langle\alpha|\alpha\rangle = \langle\beta|\beta\rangle = 1$,
$\langle\alpha|\beta\rangle = 0$ untuk setiap elektron. \omterm{def:b3:quantum-model-systems:operator}{Operator} dipol tidak
menyentuh spin, sehingga momen transisinya memuat faktor nol ini:
$T_1 \leftarrow S_0$ tidak mempunyai intensitas kecuali jika \omterm{def:b3:many-electron-atoms:spin-orbit}{kopling
spin--orbit} mencampurkan keadaan-keadaannya.
\end{solution}

\begin{solution}{pb:b3:electronic-spectroscopy:1}
\textbf{1.} $A = 5700 \times 1 \times 1.0\times10^{-4} = 0.57$.
\textbf{2.} $1 - 10^{-0.57} = 0.73$.
\textbf{3.} $\varepsilon$ sebesar beberapa ribu: transisi $\pi\to\pi^*$ yang
diizinkan, dengan kekuatan sedang.
\textbf{4.} $1239.8/349 = \qty{3.55}{eV}$.
\textbf{5.} Larutan itu hanya menyerap di ultraviolet; cahaya tampak diteruskan.
\textbf{6.} Agar cahaya yang diserap tetap sebanding dengan konsentrasi dan
untuk menghindari serapan ulang cahaya yang dipancarkan (efek filter dalam).
\textbf{7.} $S_0 \to S_1$ (serapan, lalu \omterm{def:b3:electronic-spectroscopy:radiationless}{relaksasi vibrasi}); dari $S_1$:
\omterm{def:b3:electronic-spectroscopy:luminescence}{fluoresensi}, \omterm{def:b3:electronic-spectroscopy:radiationless}{konversi internal}, \omterm{def:b3:electronic-spectroscopy:radiationless}{lintas antarsistem} ke $T_1$.
\textbf{8.} $k_r = 0.546/19 \times 10^{-9} = \qty{2.87e7}{s^{-1}}$; $\tau_r = \qty{35}{ns}$.
\textbf{9.} $(1 - 0.546)/\tau_0 = \qty{2.39e7}{s^{-1}}$.
\textbf{10.} $28653 - 22222 = \qty{6431}{cm^{-1}}$.
\textbf{11.} Emisi berawal dari $S_1$ yang telah berelaksasi dan berakhir pada
tingkat-tingkat $S_0$ yang tereksitasi secara vibrasi, setelah molekul dan
pelarutnya berelaksasi: fotonnya membawa energi yang lebih kecil, di sini
cukup kecil untuk jatuh di daerah tampak.
\textbf{12.} $\Phi_F \times (349/450) = 0.42$: 42\,\% dari energi yang diserap.
\textbf{13.} Titik-titiknya naik 0.594 untuk setiap \qty{0.010}{mol/L}: garis
lurus yang melalui 1.
\textbf{14.} Kuadrat terkecil melalui kelima titik: kemiringan
$K_{SV} = \qty{59.4}{L/mol}$, perpotongan 1.000.
\textbf{15.} $k_q = K_{SV}/\tau_0 = \qty{3.1e9}{L.mol^{-1}.s^{-1}}$.
\textbf{16.} $\tau = \tau_0/(1 + 59.4 \times 0.040) = \qty{5.6}{ns}$.
\textbf{17.} Ukurlah umurnya: untuk \omterm{def:b3:electronic-spectroscopy:quencher}{pemadaman dinamis}, $\tau_0/\tau$ jatuh pada
garis yang sama dengan $I_0/I$.
\textbf{18.} Air tonik tidak mengandung klorida yang dapat memadamkannya, dan
cukup asam untuk menjaga kinina tetap terprotonasi, yaitu bentuknya yang
berfluoresensi.
\textbf{19.} $k_d = 8 \times 8.314 \times 298/(3 \times 0.890\times10^{-3}) =
\qty{7.4e6}{m^3.mol^{-1}.s^{-1}} = \qty{7.4e9}{L.mol^{-1}.s^{-1}}$.
\textbf{20.} $k_q$ sedikit kurang dari setengah $k_d$.
\textbf{21.} Sekitar 42\,\%.
\textbf{22.} Lebih lambat: $k_d \propto 1/\eta$, dan $k_q$ hanya dapat turun
bersamanya.
\textbf{23.} \textbf{$k_q/k_d = 0.42$}: klorida memadamkan kinina pada laju
hampir setengah laju pertemuan yang dikendalikan difusi.
\end{solution}
