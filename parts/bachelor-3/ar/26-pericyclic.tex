\section{أربع عائلات}

\begin{definition}[التفاعل المحيطي الحلقي]\label{def:b3:pericyclic:pericyclic}
\emph{التفاعل المحيطي الحلقي}\index{تفاعل محيطي حلقي} تفاعل متضافر
حالته الانتقالية ترتيب حلقي من المدارات المتآثرة، تتكوّن فيه
كل الروابط وتنكسر حول الحلقة في الوقت نفسه.
\end{definition}

\begin{definition}[عائلات التفاعلات المحيطية الحلقية]\label{def:b3:pericyclic:families}
في \emph{التفاعل الحلقي الكهربائي}\index{تفاعل حلقي كهربائي} ينغلق بولي إين
مترافق في حلقة بتكوين رابطة $\sigma$ بين طرفيه، أو
العكس. وفي \emph{الإضافة الحلقية}\index{إضافة حلقية}، يتحد نظاما $\pi$
في حلقة بتكوين رابطتي $\sigma$؛ والإضافة الحلقية $[m + n]$ تصل مكوّنين
من $m$ و$n$ ذرة (أو إلكترونًا). وفي \emph{إعادة الترتيب
السيغماتروبية}\index{إعادة ترتيب سيغماتروبية} تهاجر رابطة $\sigma$ على امتداد نظام $\pi$؛
وفي الانزياح $[i, j]$ تنتقل الرابطة من الذرتين 1 و1 إلى الذرتين $i$ و$j$
في الشظيتين. و\emph{التفاعل الخيليتروبي}\index{تفاعل خيليتروبي}
يكوّن رابطتين إلى الذرة نفسها أو يكسرهما، كما حين يغادر ثنائي أكسيد الكبريت
السلفولين. وفي \emph{تفاعل الإين}\index{تفاعل الإين} يُضاف ألكين
ذو هيدروجين أليلي إلى رابطة $\pi$، فينتقل الهيدروجين معه.
\end{definition}

\begin{definition}[أحادي الوجه ومتعاكس الوجهين]\label{def:b3:pericyclic:faciality}
يتفاعل المكوّن \emph{على نحو أحادي الوجه}\index{أحادي الوجه} حين تتكوّن رابطتاه
الجديدتان، أو تنكسر رابطتاه القديمتان، على الوجه نفسه من نظامه $\pi$ (أو مع
الاحتفاظ بالتشكيل عند رابطة $\sigma$)، و\emph{على نحو متعاكس الوجهين}\index{متعاكس الوجهين} حين تكون
على وجهين متقابلين (أو مع انقلاب التشكيل).
\end{definition}

\section{التفاعلات الحلقية الكهربائية}

\begin{definition}[متساير الدوران ومتعاكس الدوران]\label{def:b3:pericyclic:rotation-modes}
في انغلاق حلقي كهربائي تدور المجموعتان الطرفيتان حول الروابط
التي تصلهما بالسلسلة. ويكون الدوران \emph{متسايرًا}\index{دوران متساير}
حين تدوران كلتاهما في الاتجاه نفسه (كلتاهما مع عقارب الساعة كما يُرى على امتداد السلسلة)، 
و\emph{متعاكسًا}\index{دوران متعاكس} حين تدوران في اتجاهين متعاكسين.
\end{definition}

\begin{theorem}[التفاعلات الحلقية الكهربائية]\label{thm:b3:pericyclic:electrocyclic}
\omterm{def:b3:pericyclic:families}{التفاعل الحلقي الكهربائي} الحراري لبولي إين ذي $4n$ إلكترونًا $\pi$ \omterm{def:b3:pericyclic:rotation-modes}{متساير
الدوران}، ولبولي إين ذي $4n + 2$ إلكترونًا $\pi$ \omterm{def:b3:pericyclic:rotation-modes}{متعاكس الدوران}.
\end{theorem}

\begin{proof}
تتكوّن الرابطة $\sigma$ الجديدة من المدارين p الطرفيين للمدار HOMO، الذي يحمل في
التفاعل الحراري الإلكترونات الأعلى طاقة؛ ولا تكون رابطة إلا إذا كان
الفصّان اللذان يتقابلان بعد الدوران من الإشارة نفسها. وبحسب صيغة هوكل
(\cref{ch:b3:solid-state}) تكون معاملات المدار $k$ لسلسلة من $N$ ذرة
متناسبة مع $\sin(jk\pi/(N + 1))$، فيكون للطرفين، $j = 1$ و$j = N$،
المعاملان $\sin(k\pi/(N + 1))$ و$\sin(Nk\pi/(N + 1)) = (-1)^{k+1}\sin(k\pi/(N + 1))$: الإشارة نفسها من أجل $k$ فردي،
وإشارتان متعاكستان من أجل $k$ زوجي. ومع $N$ ذرة و$N$ إلكترونًا يكون HOMO هو
$k = N/2$. ومن أجل $N = 4n$، $k = 2n$ زوجي: فللفصين العلويين للطرفين إشارتان
متعاكستان، ويجب أن يدور الطرفان في الاتجاه نفسه، كل منهما يأتي بفصه ذي
الإشارة الموافقة إلى الداخل: \omterm{def:b3:pericyclic:rotation-modes}{دوران متساير}. ومن أجل $N = 4n + 2$، $k = 2n + 1$ فردي: فللفصين العلويين
الإشارة نفسها، وتدويرهما أحدهما نحو الآخر \omterm{def:b3:pericyclic:rotation-modes}{دوران متعاكس}.
\end{proof}

\omterm{def:b3:pericyclic:rotation-modes}{بدوران متساير} إلى \textit{trans}-3،4-ثنائي ميثيل سيكلوبيوتين؛ وفي الواقع فإن

\omterm{def:b3:pericyclic:rotation-modes}{بدوران متعاكس} إلى \textit{cis}-5،6-ثنائي ميثيل سيكلوهكسا-1،3-ديين.

\begin{proposition}[التفاعلات الحلقية الكهربائية الضوئية]\label{prop:b3:pericyclic:photo-electrocyclic}
تحت الضوء تنعكس القواعد: $4n$ إلكترونًا، \omterm{def:b3:pericyclic:rotation-modes}{متعاكس الدوران}؛ $4n + 2$، \omterm{def:b3:pericyclic:rotation-modes}{متساير الدوران}.
\end{proposition}

\begin{proof}
امتصاص فوتون يرقّي إلكترونًا من HOMO $k = N/2$ إلى LUMO $k = N/2 + 1$
(\cref{ch:b3:photochemistry})، الذي يصير الآن أعلى مدار مشغول 
ويتحكم في الطرفين. وزوجية $k$ فيه معاكسة، فتنعكس الإشارة النسبية
لمعاملاته الطرفية، ومعها اتجاه الدوران.
\end{proof}

\begin{method}[التنبؤ بالكيمياء الفراغية لتفاعل حلقي كهربائي]\label{met:b3:pericyclic:predict-stereo}
\begin{enumerate}
  \item عُدّ إلكترونات $\pi$ في البولي إين المفتوح السلسلة ($4n$ أو $4n + 2$)؛ وسجّل الحرارة أو
        الضوء.
  \item اقرأ النمط: حراريًا $4n$ متساير، و$4n + 2$ متعاكس؛ وضوئيًا بالعكس.
  \item ارسم الطرفين بمستبدلاتهما، في التشكُّل الذي
        ينغلق (s-cis)؛ ودوّرهما بزاوية 90° وفق النمط المختار.
  \item اقرأ هل ينتهي المستبدلان على الوجه نفسه من الحلقة
        (\textit{cis}) أو على وجهين متقابلين (\textit{trans}). ومن أجل انفتاح حلقة، طبّق التحليل
        نفسه في الاتجاه المعاكس.
\end{enumerate}
\end{method}

\section{مخططات ترابط المدارات}

تنظر حجة وودوارد وهوفمان الأصلية في كل المدارات، لا في
HOMO وحده. فعلى امتداد طريق \omterm{def:b3:pericyclic:rotation-modes}{متساير الدوران} يحتفظ الجزيء بمحور $C_2$؛ وعلى امتداد
طريق \omterm{def:b3:pericyclic:rotation-modes}{متعاكس الدوران}، \omterm{def:b3:point-groups:mirror}{بمستوي مرآة} $\sigma$. وكل مدار من مدارات المتفاعل 
والناتج متناظر (S) أو ضد متناظر (A) بالنسبة إلى العنصر
المحفوظ.

\begin{definition}[مخطط ترابط المدارات]\label{def:b3:pericyclic:orbital-correlation}
\emph{مخطط ترابط المدارات}\index{مخطط ترابط المدارات} يصل
كل مدار من مدارات المتفاعل بمدار الناتج ذي التناظر نفسه،
بالنسبة إلى \omterm{def:b3:point-groups:operation}{عناصر التناظر} المحفوظة على امتداد طريق التفاعل، آخذًا إياها
بترتيب الطاقة: أدنى S مع أدنى S، وهكذا.
\end{definition}

\begin{proposition}[حفظ تناظر المدارات]\label{prop:b3:pericyclic:correlation}
إذا ترابط كل مدار مشغول في المتفاعل مع مدار مشغول في
الناتج، كان التفاعل الحراري مسموحًا؛ وإذا ترابط مدار مشغول
مع مدار فارغ، كان ممنوعًا، بحاجز مرتفع.
\end{proposition}

\begin{proof}[تعليل]
على امتداد الطريق، يحتفظ المدار بتناظره وتتغير طاقته
باستمرار؛ والمدارات المتماثلة التناظر لا يمكن أن تتقاطع (فهي تمتزج وتتنافر)،
أما المختلفة التناظر فيمكنها ذلك. وحين يجب أن يصير مدار مشغول بإلكترونين في المتفاعل
مدارًا مضادًا للترابط في الناتج، تقود الحالة الأساسية
للمتفاعل إلى حالة مثارة مرتين للناتج، وترتفع طاقة الحالة الأساسية
بشدة حتى، في وقت متأخر من الطريق، تمتزج الحالات ذات التناظر نفسه:
فيكون الحاجز مرتفعًا.
\end{proof}

\begin{method}[بناء مخطط ترابط]\label{met:b3:pericyclic:correlation-diagram}
\begin{enumerate}
  \item أوجد \omterm{def:b3:point-groups:operation}{عناصر التناظر} التي تنصّف الروابط المتكوّنة والمنكسرة 
        وتُحفظ على امتداد الطريق المختار كله.
  \item عدّد المدارات المشاركة، للمتفاعل والناتج، بترتيب
        الطاقة، ووسم كلًّا منها بالحرف S أو A لكل عنصر.
  \item صِل أدنى مدار من كل نمط تناظر في جانب بأدنى مدار من
        النمط نفسه في الجانب الآخر، ثم الذي يليه، دون أن تتقاطع خطوط من
        التناظر نفسه أبدًا.
  \item افحص المدارات المشغولة: كلها إلى مشغولة، فالتفاعل مسموح؛ وواحد إلى فارغ،
        فهو ممنوع.
\end{enumerate}
\end{method}

\begin{definition}[مخطط ترابط الحالات]\label{def:b3:pericyclic:state-correlation}
\emph{مخطط ترابط الحالات}\index{مخطط ترابط الحالات} يصل
الحالات الإلكترونية للمتفاعل والناتج، المبنية كلٌّ منها من التشكيلات
المدارية والموسومة بتناظرها الإجمالي، عبر الطريق؛ والحالات ذات
التناظر نفسه لا تتقاطع.
\end{definition}

من أجل الطريق \omterm{def:b3:pericyclic:rotation-modes}{المتعاكس الدوران} الممنوع، تترابط الحالة الأساسية للبوتاديين ($\psi_1^2\psi_2^2$)
مع حالة مثارة مرتين للسيكلوبيوتين ($\sigma^2\pi^{*2}$)، لكن الحالة المثارة الأولى
($\psi_1^2\psi_2\psi_3$) تترابط مع الحالة المثارة الأولى للسيكلوبيوتين ($\sigma^2\pi\pi^*$): فانطلاقًا من
الحالة المثارة، يكون الطريق \omterm{def:b3:pericyclic:rotation-modes}{المتعاكس الدوران} منحدرًا. وهذا هو الانعكاس الضوئي
في \cref{prop:b3:pericyclic:photo-electrocyclic}، كما يُرى على الحالات.

\section{الإضافات الحلقية}

\begin{theorem}[الإضافات الحلقية]\label{thm:b3:pericyclic:cycloaddition}
\omterm{def:b3:pericyclic:families}{الإضافة الحلقية} الحرارية $[m + n]$ التي يتفاعل فيها المكوّنان كلاهما \omterm{def:b3:pericyclic:faciality}{على نحو أحادي الوجه}
مسموحة حين يكون $m + n = 4q + 2$ إلكترونًا، وممنوعة حين $m + n = 4q$؛ وتحت الضوء تنعكس
القاعدة.
\end{theorem}

\begin{proof}
تتكوّن الرابطتان الجديدتان كلتاهما بين HOMO أحد المكوّنين وLUMO
الآخر، اللذين يتقابلان بفصوصهما الطرفية؛ ومع تفاعل المكوّنين كليهما
\omterm{def:b3:pericyclic:faciality}{على نحو أحادي الوجه}، لا تكون الرابطتان كلتاهما رابطتين إلا إذا كانت للمعاملات الطرفية في
HOMO وفي LUMO الإشارة النسبية نفسها. ومن أجل مكوّن فيه $N$
إلكترونًا على $N$ ذرة، يكون HOMO عند $k = N/2$ وLUMO عند $k = N/2 + 1$؛ والمعاملات الطرفية من
الإشارة نفسها من أجل $k$ فردي، ومتعاكسة من أجل $k$ زوجي. وفي حالة [4+2]، لكل من HOMO الديين ($k = 2$)
وLUMO الإيثين ($k = 2$) إشارتان طرفيتان متعاكستان: فيتوافقان. وفي حالة
[2+2]، يلتقي HOMO أحد الإيثينين ($k = 1$، الإشارة نفسها) بمدار LUMO للآخر
($k = 2$، متعاكستان): طرف رابط، وطرف مضاد للترابط. وعمومًا يتوافق HOMO المكوّن $m$
وLUMO المكوّن $n$ حين يكون للعددين $m/2$ و$n/2 + 1$ الزوجية
نفسها، أي حين يكون $(m + n)/2$ فرديًا: $m + n = 4q + 2$. وتحت الضوء، يؤدي LUMO المشغول بإلكترون واحد في
المكوّن المثار دور HOMO فيه، فينعكس شرط الزوجية.
\end{proof}

\section{إعادات الترتيب السيغماتروبية والقاعدة العامة}

\begin{proposition}[انزياحات الهيدروجين]\label{prop:b3:pericyclic:sigmatropic-h}
الانزياح الحراري $[1, j]$ للهيدروجين على امتداد بولي إين مسموح \omterm{def:b3:pericyclic:faciality}{على نحو أحادي الوجه} حين تحوي
الحالة الانتقالية $4q + 2$ إلكترونًا ($[1,5]$)، ويجب أن يكون \omterm{def:b3:pericyclic:faciality}{متعاكس الوجهين} حين تحوي $4q$
($[1,3]$، $[1,7]$).
\end{proposition}

\begin{proof}
لنعامل الحالة الانتقالية على أنها ذرة هيدروجين (1s) تتنقل بين طرفي
جذر من $j$ ذرة فيه $j$ إلكترونًا؛ فيجسر الهيدروجين بين طرفي
المدار SOMO، $k = (j + 1)/2$. ومن أجل $j = 5$، $k = 3$ فردي، فللطرفين فصّان من الإشارة نفسها على وجه واحد،
ويستطيع المدار 1s أن يرتبط بكليهما على ذلك الوجه: \omterm{def:b3:pericyclic:faciality}{أحادي الوجه}. ومن أجل $j = 3$ و$j = 7$،
$k = 2$ و$4$ زوجيان، والفصوص ذات الإشارة نفسها على وجهين متقابلين عند الطرفين، 
ويجب أن ينتقل الهيدروجين من وجه إلى الآخر: \omterm{def:b3:pericyclic:faciality}{متعاكس الوجهين}. ولا يستطيع الانزياح $[1,3]$
أن يمتد عبر سلسلة بهذا القصر؛ أما الانزياح $[1,7]$ فيستطيع، على امتداد بولي إين حلزوني.
\end{proof}

كلايزن}\index{إعادة ترتيب كلايزن} هي إعادة ترتيب إيثر أليل فينيل، أو

\end{definition}

\begin{theorem}[قواعد وودوارد--هوفمان]\label{thm:b3:pericyclic:woodward-hoffmann}
يكون \omterm{def:b3:pericyclic:pericyclic}{التفاعل المحيطي الحلقي} الحراري مسموحًا حين يكون العدد الكلي للمكوّنات $(4q + 2)_s$ 
و$(4r)_a$ فرديًا؛ والضوئي حين يكون زوجيًا.
\end{theorem}

\begin{proof}[برهان جزئي]
من أجل كل عائلة عولجت أعلاه يتوافق العدّ مع تحليل
المدارات الحدودية: \omterm{def:b3:pericyclic:families}{فالإضافة الحلقية} الحرارية [4+2] لها المكوّنان $\pi4_s$ و$\pi2_s$، أي مكوّن واحد $(4q + 2)_s$،
فردي: مسموحة؛ والتفاعل [2+2] له $\pi2_s + \pi2_s$، أي اثنان، زوجي: ممنوع، بينما
$\pi2_s + \pi2_a$ له واحد: مسموح؛ والانفتاح \omterm{def:b3:pericyclic:rotation-modes}{المتساير الدوران} للسيكلوبيوتين هو $\sigma2_s + \pi2_a$، واحد:
مسموح؛ والانزياح \babelsublr{$[1,5]$-H} \omterm{def:b3:pericyclic:faciality}{الأحادي الوجه} هو $\sigma2_s + \pi4_s$، حيث لا يُعدّ إلا $\sigma2_s$: واحد،
مسموح. أما البرهان العام، الصالح لكل ترتيب من المكوّنات، فينتمي
إلى مقررات أكثر تقدمًا.
\end{proof}

\begin{definition}[الحالات الانتقالية العطرية ومن نمط موبيوس]\label{def:b3:pericyclic:mobius}
\emph{الحالة الانتقالية العطرية}\index{حالة انتقالية عطرية} ترتيب
حلقي من المدارات في حالة انتقالية محيطية حلقية، يستقر، كحلقة هوكل،
حين يحوي $4q + 2$ إلكترونًا. و\emph{الحالة الانتقالية من نمط
موبيوس}\index{حالة انتقالية من نمط موبيوس} فيها عدد فردي من تغيرات الإشارة بين
المدارات المتجاورة حول الحلقة (التواء واحد \omterm{def:b3:pericyclic:faciality}{متعاكس الوجهين})؛ وهي
تستقر مع $4q$ إلكترونًا.
\end{definition}

\begin{proposition}[الحالات الانتقالية العطرية]\label{prop:b3:pericyclic:aromatic-ts}
يكون \omterm{def:b3:pericyclic:pericyclic}{التفاعل المحيطي الحلقي} الحراري مسموحًا حين تكون حالته الانتقالية عطرية:
طوبولوجيا هوكل مع $4q + 2$ إلكترونًا، أو طوبولوجيا موبيوس مع $4q$ إلكترونًا.
\end{proposition}

\begin{proof}[تعليل]
يسلك الترتيب الحلقي للمدارات في الحالة الانتقالية سلوك جزيء حلقي
مترافق. فلحلقة هوكل مستوى أدنى غير منحل وأزواج
فوقه، تمتلئ بعدد $4q + 2$ من الإلكترونات (مجلد السنة الثانية)؛ وللحلقة ذات انقلاب
إشارة واحد كل مستوياتها أزواج، تمتلئ بعدد $4q$. والطبقة الممتلئة تعني
الاستقرار، \omterm{def:b3:pericyclic:mobius}{وحالة انتقالية عطرية} وحاجزًا منخفضًا. وعدد تغيرات
الإشارة هو عدد المكوّنات المتعاكسة الوجهين، فهذه القاعدة مكافئة
لقاعدة وودوارد--هوفمان.
\end{proof}

\begin{method}[رسم المكوّنات وعدّها]\label{met:b3:pericyclic:components}
\begin{enumerate}
  \item علّم الروابط المتكوّنة والمنكسرة؛ فكل نظام $\pi$ أو رابطة $\sigma$ تتغير
        مكوّن، يوسم بنوعه وعدد إلكتروناته ($\pi4$، $\pi2$، $\sigma2$، $\omega0$ من أجل
        مدار فارغ).
  \item اختر s أو a لكل مكوّن بحسب الهندسة (الوجه نفسه، أو
        وجهان متقابلان).
  \item عُدّ المكوّنات $(4q + 2)_s$ و$(4r)_a$: فردي، مسموح حراريًا؛ زوجي، مسموح
        ضوئيًا.
  \item تحقّق بعدد الإلكترونات حول الحلقة: هوكل مع $4q + 2$،
        وموبيوس مع $4q$.
\end{enumerate}
\end{method}

\begin{inthelab}[إعادة ترتيب كلايزن]
يُسخَّن إيثر أليل أريل مع التقطير الارتدادي، تحت النيتروجين، في مذيب
عالي الغليان مثل 1،2-ثنائي كلوروبنزين، أو دون مذيب، عند نحو \qty{200}{\celsius}، 
ويُتابَع التفاعل بكروماتوغرافيا الطبقة الرقيقة: فالمركب \textit{ortho}-أليل فينول المتكوّن
أكثر قطبية من الإيثر. ثم يُستخلص الفينول في محلول مائي من هيدروكسيد
الصوديوم، ويُفصل عن الشوائب المتعادلة، ويُستعاد بالتحميض.
\end{inthelab}

لخطوات حلقية كهربائية في أثناء تخليقه الفيتامين \babelsublr{B$_{12}$}، ورولد هوفمان
قواعد حفظ تناظر المدارات. وكان كينيتشي فوكوي قد قدّم
المدارات الحدودية سنة 1952؛ وتقاسم فوكوي وهوفمان جائزة نوبل في الكيمياء
لسنة 1981. أما وودوارد، الذي نال جائزة سنة 1965 عن تخليقاته، فقد توفي سنة
1979.

\section{مسألة: ضوء الشمس والجلد والفيتامين D}

\begin{problem}[{مسألة نهاية الأسبوع --- ضوء الشمس والجلد والفيتامين D: الانفتاح
الحلقي الضوئي للمركب 7-ديهيدروكوليسترول، وانزياح الهيدروجين الحراري،
والنواتج الجانبية، والزمن الذي يستغرقه الجسم لإتمام العمل}]
\label{pb:b3:pericyclic:1}
معطيات المسألة: عند \qty{37}{\celsius} يكون تحويل طليعة الفيتامين \babelsublr{D$_3$} إلى الفيتامين \babelsublr{D$_3$}
من الرتبة الأولى مع $k = \qty{2.3e-2}{h^{-1}}$ (ويُهمل عكسه)؛ وطاقة تنشيطه
\qty{85}{kJ/mol}.

\medskip\noindent\textbf{الجزء الأول --- الانفتاح الحلقي.}
\begin{enumerate}
  \item أي رابطة في 7-ديهيدروكوليسترول تنكسر، وأي نظام $\pi$ يتكوّن؟
  \item كم إلكترونًا يشارك، وما عائلة التفاعل؟
  \item لماذا يحتاج إلى الضوء؟
  \item هل الانفتاح الضوئي \omterm{def:b3:pericyclic:rotation-modes}{متساير الدوران} أم \omterm{def:b3:pericyclic:rotation-modes}{متعاكس الدوران}؟ علّل ذلك
        بالمدارات.
  \item لماذا يجب أن تكون الرابطة المضاعفة المركزية الجديدة في طليعة الفيتامين \babelsublr{D$_3$} من النوع \textit{Z}؟
  \item لماذا لا يمكن أن يحدث الانفتاح نفسه حراريًا في الظلام عند درجة حرارة الجسم؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثاني --- انزياح الهيدروجين.}
\begin{enumerate}[resume]
  \item سمِّ الذرتين اللتين ينتقل الهيدروجين بينهما، ونوع الانزياح.
  \item كم إلكترونًا يشارك في الحالة الانتقالية؟
  \item عُدّ المكوّنات وقرّر هل التفاعل الحراري مسموح \omterm{def:b3:pericyclic:faciality}{على نحو
        أحادي الوجه} أم \omterm{def:b3:pericyclic:faciality}{متعاكس الوجهين}.
  \item لماذا يكون الانزياح \omterm{def:b3:pericyclic:faciality}{المتعاكس الوجهين} ممكنًا هندسيًا هنا، بينما الانزياح $[1,3]$
        غير ممكن؟
  \item هل الحالة الانتقالية من نمط هوكل أم موبيوس، وهل هي عطرية؟
  \item لماذا لا يحتاج الانزياح إلى الضوء؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الثالث --- النواتج الجانبية.}
\begin{enumerate}[resume]
  \item يمكن أن تنغلق طليعة الفيتامين \babelsublr{D$_3$} من جديد تحت الضوء. بأي نمط، ولماذا يمكن أن تعطي
        متماكبًا فراغيًا حلقيًا للمركب 7-ديهيدروكوليسترول (اللوميستيرول)؟
  \item يمكن للضوء أيضًا أن يحوّل الرابطة المضاعفة المركزية \textit{Z} إلى \textit{E} (التاكيستيرول). لماذا
        لا يستطيع التاكيستيرول أن يخضع للانزياح \babelsublr{$[1,7]$-H}؟
  \item لماذا لا ينتج ضوء الشمس المطوّل مزيدًا من الفيتامين D بلا حد؟
  \item لماذا يكون الانفتاح الحلقي سريعًا وانزياح الهيدروجين بطيئًا؟
\end{enumerate}

\medskip\noindent\textbf{الجزء الرابع --- الحركية عند درجة حرارة الجسم.}
\begin{enumerate}[resume]
  \item احسب عمر النصف لطليعة الفيتامين \babelsublr{D$_3$} عند \qty{37}{\celsius}.
  \item احسب النسبة المحوَّلة في يوم واحد.
  \item احسب الزمن اللازم لتحويل 90\,\%.
  \item احسب $k$ عند \qty{20}{\celsius}.
  \item احسب الزمن اللازم لتحويل 90\,\% عند \qty{20}{\celsius}.
  \item لماذا يفيد أن يكون الجلد دافئًا؟
  \item لماذا تُصنع مكملات الفيتامين D بتشعيع ستيرول ثم تدفئته،
        بدل تخليق كلاسيكي؟
  \item اذكر النتيجة: الزمن اللازم لتحويل 90\,\% من طليعة الفيتامين \babelsublr{D$_3$} عند درجة حرارة الجسم.
\end{enumerate}
\end{problem}
