\section{Empat golongan}

\begin{definition}[Reaksi perisiklik]\label{def:b3:pericyclic:pericyclic}
Yang disebut \emph{reaksi perisiklik}\index{reaksi perisiklik} adalah reaksi serempak
yang keadaan transisinya berupa susunan siklik orbital-orbital yang berinteraksi, tempat semua
ikatan terbentuk dan putus di sekeliling cincin pada saat yang sama.
\end{definition}

\begin{definition}[Golongan-golongan reaksi perisiklik]\label{def:b3:pericyclic:families}
Dalam \emph{reaksi elektrosiklik}\index{reaksi elektrosiklik}, sebuah poliena
terkonjugasi menutup menjadi cincin dengan membentuk ikatan $\sigma$ di antara kedua ujungnya, atau
sebaliknya. Dalam \emph{sikloadisi}\index{sikloadisi}, dua sistem $\pi$ bergabung
menjadi cincin dengan membentuk dua ikatan $\sigma$; sikloadisi $[m + n]$ menggabungkan komponen-komponen
yang terdiri atas $m$ dan $n$ atom (atau elektron). Dalam \emph{penataan ulang
sigmatropik}\index{penataan ulang sigmatropik}, sebuah ikatan $\sigma$ bermigrasi sepanjang sistem $\pi$;
dalam pergeseran $[i, j]$, ikatan itu berpindah dari atom 1 dan 1 ke atom $i$ dan $j$
kedua fragmen. Dalam \emph{reaksi keletropik}\index{reaksi keletropik},
dua ikatan dengan atom yang sama terbentuk atau putus, seperti ketika belerang dioksida
lepas dari sebuah sulfolena. Dalam \emph{reaksi ena}\index{reaksi ena}, sebuah alkena
yang memiliki hidrogen alilik teradisi pada ikatan $\pi$, sementara hidrogen itu ikut berpindah.
\end{definition}

\begin{definition}[Suprafasial dan antarafasial]\label{def:b3:pericyclic:faciality}
Sebuah komponen bereaksi secara \emph{suprafasial}\index{suprafasial} jika kedua ikatan barunya
terbentuk, atau kedua ikatan lamanya putus, pada muka yang sama sistem $\pi$-nya (atau dengan
retensi pada ikatan $\sigma$), dan secara \emph{antarafasial}\index{antarafasial} jika kedua ikatan itu
berada pada muka yang berlawanan (atau dengan inversi).
\end{definition}

\section{Reaksi elektrosiklik}

\begin{definition}[Konrotatori dan disrotatori]\label{def:b3:pericyclic:rotation-modes}
Dalam penutupan cincin elektrosiklik, kedua gugus ujung berputar terhadap ikatan-ikatan
yang menghubungkannya dengan rantai. Rotasinya disebut \emph{konrotatori}\index{konrotatori}
jika keduanya berputar ke arah yang sama (keduanya searah jarum jam dilihat sepanjang rantai), dan
\emph{disrotatori}\index{disrotatori} jika keduanya berputar ke arah yang berlawanan.
\end{definition}

\begin{theorem}[Reaksi elektrosiklik]\label{thm:b3:pericyclic:electrocyclic}
\omterm{def:b3:pericyclic:families}{Reaksi elektrosiklik} termal sebuah poliena dengan $4n$ elektron $\pi$ bersifat
\omterm{def:b3:pericyclic:rotation-modes}{konrotatori}, dan dengan $4n + 2$ elektron $\pi$ bersifat \omterm{def:b3:pericyclic:rotation-modes}{disrotatori}.
\end{theorem}

\begin{proof}
Ikatan $\sigma$ yang baru terbentuk dari orbital p ujung-ujung HOMO, yang dalam
reaksi termal memuat elektron-elektron berenergi tertinggi; ikatan itu bersifat ikatan hanya jika
cuping-cuping yang berputar sehingga saling berhadapan bertanda sama. Menurut rumus Hückel
(\cref{ch:b3:solid-state}), koefisien orbital $k$ rantai $N$ atom
sebanding dengan $\sin(jk\pi/(N + 1))$, sehingga kedua ujung, $j = 1$ dan $j = N$, memiliki
koefisien $\sin(k\pi/(N + 1))$ dan $\sin(Nk\pi/(N + 1)) = (-1)^{k+1}\sin(k\pi/(N + 1))$: bertanda sama untuk $k$ ganjil,
berlawanan tanda untuk $k$ genap. Dengan $N$ atom dan $N$ elektron, HOMO adalah
$k = N/2$. Untuk $N = 4n$, $k = 2n$ genap: cuping-cuping atas kedua ujung berlawanan
tanda, dan kedua ujung harus berputar ke arah yang sama, masing-masing membawa cupingnya yang
bertanda cocok ke dalam: \omterm{def:b3:pericyclic:rotation-modes}{konrotatori}. Untuk $N = 4n + 2$, $k = 2n + 1$ ganjil: cuping-cuping atas
bertanda sama, dan memutarnya agar saling berhadapan berarti \omterm{def:b3:pericyclic:rotation-modes}{disrotatori}.
\end{proof}

\begin{proposition}[Reaksi elektrosiklik fotokimia]\label{prop:b3:pericyclic:photo-electrocyclic}
Di bawah cahaya, aturan-aturan itu terbalik: $4n$ elektron, \omterm{def:b3:pericyclic:rotation-modes}{disrotatori}; $4n + 2$, \omterm{def:b3:pericyclic:rotation-modes}{konrotatori}.
\end{proposition}

\begin{proof}
Penyerapan sebuah foton mempromosikan sebuah elektron dari HOMO $k = N/2$ ke LUMO $k = N/2 + 1$
(\cref{ch:b3:photochemistry}), yang kini menjadi orbital terisi tertinggi dan
mengendalikan ujung-ujungnya. $k$-nya berparitas berlawanan, sehingga tanda relatif koefisien
ujungnya terbalik, dan bersamanya arah rotasinya.
\end{proof}

\begin{method}[Meramalkan stereokimia reaksi elektrosiklik]\label{met:b3:pericyclic:predict-stereo}
\begin{enumerate}
  \item Hitunglah elektron $\pi$ poliena rantai terbuka itu ($4n$ atau $4n + 2$); catatlah panas atau
        cahaya.
  \item Bacalah modusnya: termal $4n$ kon, $4n + 2$ dis; fotokimia terbalik.
  \item Gambarlah ujung-ujungnya beserta substituennya, dalam konformasi yang
        menutup (s-cis); putarlah keduanya 90° menurut modus yang dipilih.
  \item Bacalah apakah kedua substituen berakhir pada muka cincin yang sama
        (\textit{cis}) atau pada muka yang berlawanan (\textit{trans}). Untuk pembukaan cincin, jalankan analisis
        yang sama secara mundur.
\end{enumerate}
\end{method}

\section{Diagram korelasi orbital}

Argumen asli Woodward dan Hoffmann memperhatikan semua orbital, tidak hanya
HOMO. Sepanjang jalur \omterm{def:b3:pericyclic:rotation-modes}{konrotatori}, molekul mempertahankan sumbu $C_2$; sepanjang
jalur \omterm{def:b3:pericyclic:rotation-modes}{disrotatori}, \omterm{def:b3:point-groups:mirror}{bidang cermin} $\sigma$. Setiap orbital pereaksi dan
produk bersifat simetris (S) atau antisimetris (A) terhadap \omterm{def:b3:point-groups:operation}{unsur simetri} yang
dipertahankan.

\begin{definition}[Diagram korelasi orbital]\label{def:b3:pericyclic:orbital-correlation}
Yang disebut \emph{diagram korelasi orbital}\index{diagram korelasi orbital} adalah diagram yang menghubungkan
setiap orbital pereaksi dengan orbital produk yang simetrinya sama,
terhadap \omterm{def:b3:point-groups:operation}{unsur-unsur simetri} yang dipertahankan sepanjang jalur reaksi, dengan mengambilnya
menurut urutan energi: S terendah dengan S terendah, dan seterusnya.
\end{definition}

\begin{proposition}[Kekekalan simetri orbital]\label{prop:b3:pericyclic:correlation}
Jika setiap orbital yang terisi dalam pereaksi berkorelasi dengan orbital yang terisi dalam
produk, reaksi termalnya diizinkan; jika sebuah orbital terisi berkorelasi
dengan orbital kosong, reaksi itu terlarang, dengan penghalang yang tinggi.
\end{proposition}

\begin{proof}[Argumen]
Sepanjang jalur, sebuah orbital mempertahankan simetrinya dan energinya berubah
secara kontinu; orbital-orbital yang simetrinya sama tidak dapat bersilangan (keduanya bercampur dan saling tolak),
sedangkan orbital-orbital yang simetrinya berbeda dapat. Ketika sebuah orbital pereaksi yang terisi ganda
harus menjadi orbital antiikatan produk, keadaan dasar
pereaksi menuju keadaan tereksitasi ganda produk, dan energi keadaan dasar
naik tajam sampai, di bagian akhir jalur, keadaan-keadaan yang simetrinya sama
bercampur: penghalangnya tinggi.
\end{proof}

\begin{method}[Membangun diagram korelasi]\label{met:b3:pericyclic:correlation-diagram}
\begin{enumerate}
  \item Carilah \omterm{def:b3:point-groups:operation}{unsur-unsur simetri} yang membagi dua ikatan-ikatan yang terbentuk dan putus serta
        dipertahankan sepanjang jalur yang dipilih.
  \item Daftarlah orbital-orbital yang terlibat, dari pereaksi dan produk, menurut urutan
        energi, dan berilah label S atau A untuk setiap unsur.
  \item Hubungkan orbital terendah setiap tipe simetri di satu sisi dengan orbital terendah
        bertipe sama di sisi lain, lalu yang berikutnya, tanpa pernah menyilangkan dua garis yang
        simetrinya sama.
  \item Periksalah orbital-orbital terisi: semuanya menuju orbital terisi, diizinkan; satu menuju orbital kosong,
        terlarang.
\end{enumerate}
\end{method}

\begin{definition}[Diagram korelasi keadaan]\label{def:b3:pericyclic:state-correlation}
Yang disebut \emph{diagram korelasi keadaan}\index{diagram korelasi keadaan} adalah diagram yang menghubungkan
keadaan-keadaan elektronik pereaksi dan produk, yang masing-masing dibangun dari konfigurasi-konfigurasi
orbital dan diberi label menurut simetri keseluruhannya, melalui jalur reaksi; keadaan-keadaan yang
simetrinya sama tidak bersilangan.
\end{definition}

Untuk jalur \omterm{def:b3:pericyclic:rotation-modes}{disrotatori} yang terlarang, keadaan dasar butadiena ($\psi_1^2\psi_2^2$)
berkorelasi dengan keadaan tereksitasi ganda siklobutena ($\sigma^2\pi^{*2}$), tetapi keadaan tereksitasi
pertama ($\psi_1^2\psi_2\psi_3$) berkorelasi dengan keadaan tereksitasi pertama siklobutena ($\sigma^2\pi\pi^*$): dari
keadaan tereksitasi, jalur \omterm{def:b3:pericyclic:rotation-modes}{disrotatori} menurun. Itulah pembalikan fotokimia
pada \cref{prop:b3:pericyclic:photo-electrocyclic}, yang dilihat pada keadaan-keadaan.

\section{Sikloadisi}

\begin{theorem}[Sikloadisi]\label{thm:b3:pericyclic:cycloaddition}
\omterm{def:b3:pericyclic:families}{Sikloadisi} $[m + n]$ termal yang kedua komponennya bereaksi secara \omterm{def:b3:pericyclic:faciality}{suprafasial}
diizinkan jika $m + n = 4q + 2$ elektron, dan terlarang jika $m + n = 4q$; di bawah cahaya,
aturan itu terbalik.
\end{theorem}

\begin{proof}
Kedua ikatan baru terbentuk antara HOMO satu komponen dan LUMO komponen
lainnya, yang saling berhadapan melalui cuping-cuping ujungnya; jika keduanya bereaksi
secara \omterm{def:b3:pericyclic:faciality}{suprafasial}, kedua ikatan bersifat ikatan hanya jika koefisien ujung
HOMO dan LUMO memiliki tanda relatif yang sama. Untuk komponen dengan $N$
elektron pada $N$ atom, HOMO memiliki $k = N/2$ dan LUMO $k = N/2 + 1$; koefisien ujungnya
bertanda sama untuk $k$ ganjil, berlawanan untuk $k$ genap. Pada kasus [4+2], HOMO diena ($k = 2$)
dan LUMO etena ($k = 2$) sama-sama memiliki tanda ujung yang berlawanan: keduanya cocok. Pada
kasus [2+2], HOMO satu etena ($k = 1$, tanda sama) bertemu dengan LUMO etena yang lain
($k = 2$, berlawanan): satu ujung bersifat ikatan, satu antiikatan. Secara umum, HOMO komponen $m$
dan LUMO komponen $n$ cocok jika $m/2$ dan $n/2 + 1$ berparitas
sama, yaitu jika $(m + n)/2$ ganjil: $m + n = 4q + 2$. Di bawah cahaya, LUMO yang terisi tunggal pada
komponen yang tereksitasi berperan sebagai HOMO-nya, dan syarat paritasnya
terbalik.
\end{proof}

\begin{definition}[Sikloadisi dipolar-1,3]\label{def:b3:pericyclic:dipolar}
Yang disebut \emph{dipol-1,3}\index{dipol-1,3} adalah sistem $\pi$ tiga atom dengan empat
elektron, yang dituliskan dengan muatan formal di ujung-ujungnya atau di tengahnya, misalnya
azida $\ce{R-N=N+=N-}$, nitril oksida, atau molekul ozon. Yang disebut \emph{sikloadisi
dipolar-1,3}\index{sikloadisi dipolar-1,3} adalah \omterm{def:b3:pericyclic:families}{sikloadisi} [3+2] dipol itu, yaitu
reaksi enam elektron, dengan alkena atau alkuna, yang memberikan cincin beranggota lima.
\end{definition}

Reaksi azida dengan alkuna terminal yang dikatalisis tembaga, yang memberikan satu
regioisomer 1,2,3-triazola, adalah reaksi penyambungan yang andal dalam kimia
klik. Ozonolisis dimulai dengan \omterm{def:b3:pericyclic:dipolar}{sikloadisi dipolar-1,3} ozon pada
alkena.

\section{Penataan ulang sigmatropik dan aturan umum}

\begin{proposition}[Pergeseran hidrogen]\label{prop:b3:pericyclic:sigmatropic-h}
Pergeseran $[1, j]$ termal sebuah hidrogen sepanjang poliena diizinkan secara \omterm{def:b3:pericyclic:faciality}{suprafasial} jika
keadaan transisinya memuat $4q + 2$ elektron ($[1,5]$) dan harus \omterm{def:b3:pericyclic:faciality}{antarafasial} jika memuat $4q$
($[1,3]$, $[1,7]$).
\end{proposition}

\begin{proof}
Perlakukan keadaan transisi sebagai atom hidrogen (1s) yang berpindah di antara ujung-ujung
radikal $j$ atom dengan $j$ elektron; hidrogen itu menjembatani kedua ujung
SOMO, $k = (j + 1)/2$. Untuk $j = 5$, $k = 3$ ganjil, ujung-ujungnya memiliki cuping bertanda sama pada satu muka,
dan orbital 1s dapat berikatan dengan keduanya pada muka itu: \omterm{def:b3:pericyclic:faciality}{suprafasial}. Untuk $j = 3$ dan $j = 7$,
$k = 2$ dan $4$ genap, cuping-cuping bertanda sama berada pada muka yang berlawanan di kedua ujung, dan
hidrogen harus berpindah dari satu muka ke muka yang lain: \omterm{def:b3:pericyclic:faciality}{antarafasial}. Pergeseran $[1,3]$
tidak dapat menjangkau rantai sependek itu; pergeseran $[1,7]$ dapat, sepanjang poliena yang berbentuk heliks.
\end{proof}

heksa-1,5-diena; sedangkan \emph{penataan ulang
Claisen}\index{penataan ulang Claisen} adalah penataan ulang serupa pada alil vinil eter, atau pada
alil aril eter, menjadi senyawa karbonil $\gamma,\delta$-tak jenuh atau \textit{orto}-alilfenol.
\end{definition}

Pergeseran [3,3] adalah proses enam elektron yang seluruhnya \omterm{def:b3:pericyclic:faciality}{suprafasial}, yang diizinkan secara termal;

\begin{theorem}[Aturan Woodward--Hoffmann]\label{thm:b3:pericyclic:woodward-hoffmann}
\omterm{def:b3:pericyclic:pericyclic}{Reaksi perisiklik} termal diizinkan jika jumlah total komponen $(4q + 2)_s$ dan
$(4r)_a$ ganjil; reaksi fotokimia diizinkan jika jumlah itu genap.
\end{theorem}

\begin{proof}[Bukti parsial]
Untuk setiap golongan yang dibahas di atas, hitungan ini sesuai dengan analisis
orbital batas: \omterm{def:b3:pericyclic:families}{sikloadisi} [4+2] termal memiliki komponen $\pi4_s$ dan $\pi2_s$, satu komponen $(4q + 2)_s$,
ganjil: diizinkan; [2+2] memiliki $\pi2_s + \pi2_s$, dua, genap: terlarang, sedangkan
$\pi2_s + \pi2_a$ memiliki satu: diizinkan; pembukaan \omterm{def:b3:pericyclic:rotation-modes}{konrotatori} siklobutena adalah $\sigma2_s + \pi2_a$, satu:
diizinkan; pergeseran $[1,5]$-H \omterm{def:b3:pericyclic:faciality}{suprafasial} adalah $\sigma2_s + \pi4_s$, dan hanya $\sigma2_s$ yang dihitung: satu,
diizinkan. Bukti umumnya, yang berlaku untuk setiap susunan komponen, termasuk dalam
mata kuliah yang lebih lanjut.
\end{proof}

\begin{definition}[Keadaan transisi aromatik dan Möbius]\label{def:b3:pericyclic:mobius}
Yang disebut \emph{keadaan transisi aromatik}\index{keadaan transisi aromatik} adalah susunan
siklik orbital-orbital dalam keadaan transisi perisiklik yang, seperti cincin Hückel,
terstabilkan jika memuat $4q + 2$ elektron. Yang disebut \emph{keadaan transisi
Möbius}\index{keadaan transisi Möbius} adalah keadaan transisi yang memiliki jumlah ganjil perubahan tanda di antara
orbital-orbital yang bertetangga di sekeliling cincin (satu puntiran \omterm{def:b3:pericyclic:faciality}{antarafasial}); keadaan itu
terstabilkan dengan $4q$ elektron.
\end{definition}

\begin{proposition}[Keadaan transisi aromatik]\label{prop:b3:pericyclic:aromatic-ts}
\omterm{def:b3:pericyclic:pericyclic}{Reaksi perisiklik} termal diizinkan jika keadaan transisinya aromatik:
topologi Hückel dengan $4q + 2$ elektron, atau topologi Möbius dengan $4q$ elektron.
\end{proposition}

\begin{proof}[Argumen]
Susunan siklik orbital-orbital dalam keadaan transisi berperilaku seperti molekul
siklik terkonjugasi. Cincin Hückel memiliki tingkat terendah yang tidak terdegenerasi dan pasangan-pasangan tingkat
di atasnya, yang terisi penuh dengan $4q + 2$ elektron (Jilid Tahun 2); cincin dengan satu
pembalikan tanda memiliki semua tingkatnya berpasangan, yang terisi penuh dengan $4q$. Kulit yang penuh berarti
stabilisasi, \omterm{def:b3:pericyclic:mobius}{keadaan transisi aromatik}, dan penghalang yang rendah. Jumlah perubahan
tanda sama dengan jumlah komponen \omterm{def:b3:pericyclic:faciality}{antarafasial}, sehingga aturan ini setara dengan
aturan Woodward--Hoffmann.
\end{proof}

\begin{method}[Menggambar dan menghitung komponen]\label{met:b3:pericyclic:components}
\begin{enumerate}
  \item Tandailah ikatan-ikatan yang terbentuk dan putus; setiap sistem $\pi$ atau ikatan $\sigma$ yang berubah adalah sebuah
        komponen, yang diberi label dengan tipe dan jumlah elektronnya ($\pi4$, $\pi2$, $\sigma2$, $\omega0$ untuk
        orbital kosong).
  \item Pilihlah s atau a untuk setiap komponen dari geometrinya (muka yang sama, atau
        muka yang berlawanan).
  \item Hitunglah komponen $(4q + 2)_s$ dan $(4r)_a$: ganjil, diizinkan secara termal; genap, diizinkan
        secara fotokimia.
  \item Periksalah silang dengan jumlah elektron di sekeliling cincin: Hückel dengan $4q + 2$,
        Möbius dengan $4q$.
\end{enumerate}
\end{method}

\begin{inthelab}[Sebuah penataan ulang Claisen]
Sebuah alil aril eter dipanaskan dengan refluks, di bawah nitrogen, dalam pelarut
bertitik didih tinggi seperti 1,2-diklorobenzena, atau tanpa pelarut, pada sekitar \qty{200}{\celsius}, dan
reaksinya diikuti dengan kromatografi lapis tipis: \textit{orto}-alilfenol yang terbentuk
lebih polar daripada eternya. Fenol itu kemudian diekstraksi ke dalam natrium
hidroksida berair, dipisahkan dari pengotor netral, dan diperoleh kembali dengan pengasaman.
\end{inthelab}

\begin{safety}{\ghs[0.62cm]{GHS02}\ghs[0.62cm]{GHS04}\ghs[0.62cm]{GHS08}}
% ledger: ghs4:butadiene
Buta-1,3-diena, diena yang lazim: gas bertekanan yang sangat mudah menyala, dapat menyebabkan
cacat genetik dan kanker. Zat itu hanya ditangani dalam sistem tertutup; dalam pengajaran,
sulfolena padat, yang melepaskannya dalam \omterm{def:b3:pericyclic:families}{reaksi keletropik} jika dipanaskan, dipakai
sebagai penggantinya dalam peralatan tertutup.
\end{safety}

\begin{history}[Simetri orbital]
Pada 1965 Robert Woodward, yang telah menemukan stereokimia tahap-tahap elektrosiklik yang membingungkan
ketika menyintesis vitamin B$_{12}$, dan Roald Hoffmann menerbitkan
aturan kekekalan simetri orbital. Kenichi Fukui telah memperkenalkan
orbital batas pada 1952; Fukui dan Hoffmann berbagi Hadiah Nobel Kimia
1981. Woodward, yang telah menerima hadiah 1965 untuk sintesis-sintesisnya, wafat pada
1979.
\end{history}

\section{Soal: Sinar Matahari, Kulit, dan Vitamin D}

\begin{problem}[{Soal akhir pekan --- sinar matahari, kulit, dan vitamin D:
pembukaan cincin fotokimia 7-dehidrokolesterol, pergeseran hidrogen termal,
produk-produk samping, dan berapa lama tubuh menyelesaikan pekerjaan itu}]
\label{pb:b3:pericyclic:1}
Data soal: pada \qty{37}{\celsius}, konversi pravitamin D$_3$ menjadi vitamin D$_3$ berorde
satu dengan $k = \qty{2.3e-2}{h^{-1}}$ (reaksi kebalikannya diabaikan); energi aktivasinya
\qty{85}{kJ/mol}.

\medskip\noindent\textbf{Bagian I --- Pembukaan cincin.}
\begin{enumerate}[beginpenalty=10000]
  \item Ikatan 7-dehidrokolesterol manakah yang putus, dan sistem $\pi$ apakah yang terbentuk?
  \item Berapa banyak elektron yang terlibat, dan termasuk golongan reaksi apakah?
  \item Mengapa reaksi itu memerlukan cahaya?
  \item Apakah pembukaan fotokimia itu \omterm{def:b3:pericyclic:rotation-modes}{konrotatori} atau \omterm{def:b3:pericyclic:rotation-modes}{disrotatori}? Jelaskan dengan orbitalnya.
  \item Mengapa ikatan rangkap tengah yang baru pada pravitamin D$_3$ harus \textit{Z}?
  \item Mengapa pembukaan itu tidak terjadi secara termal dalam gelap pada suhu tubuh?
\end{enumerate}

\medskip\noindent\textbf{Bagian II --- Pergeseran hidrogen.}
\begin{enumerate}[resume, beginpenalty=10000]
  \item Sebutkan atom-atom tempat hidrogen itu berpindah, dan tipe pergeserannya.
  \item Berapa banyak elektron yang terlibat dalam keadaan transisi?
  \item Hitunglah komponen-komponennya dan tentukan apakah reaksi termalnya diizinkan
        secara \omterm{def:b3:pericyclic:faciality}{suprafasial} atau \omterm{def:b3:pericyclic:faciality}{antarafasial}.
  \item Mengapa pergeseran \omterm{def:b3:pericyclic:faciality}{antarafasial} mungkin secara geometris di sini, tetapi $[1,3]$ tidak?
  \item Apakah keadaan transisinya Hückel atau Möbius, dan apakah aromatik?
  \item Mengapa pergeseran itu tidak memerlukan cahaya?
\end{enumerate}

\medskip\noindent\textbf{Bagian III --- Produk samping.}
\begin{enumerate}[resume, beginpenalty=10000]
  \item Pravitamin D$_3$ dapat menutup kembali di bawah cahaya. Dengan modus apakah, dan mengapa penutupan itu dapat memberikan
        stereoisomer cincin 7-dehidrokolesterol (lumisterol)?
  \item Cahaya juga dapat mengubah ikatan rangkap tengah \textit{Z} menjadi \textit{E} (takisterol). Mengapa
        takisterol tidak dapat mengalami pergeseran $[1,7]$-H?
  \item Mengapa sinar matahari yang lama tidak terus menambah vitamin D?
  \item Mengapa pembukaan cincin berlangsung cepat tetapi pergeseran hidrogen lambat?
\end{enumerate}

\medskip\noindent\textbf{Bagian IV --- Kinetika pada suhu tubuh.}
\begin{enumerate}[resume, beginpenalty=10000]
  \item Hitunglah waktu paruh pravitamin D$_3$ pada \qty{37}{\celsius}.
  \item Hitunglah fraksi yang terkonversi dalam satu hari.
  \item Hitunglah waktu untuk mengonversi 90\,\%.
  \item Hitunglah $k$ pada \qty{20}{\celsius}.
  \item Hitunglah waktu untuk mengonversi 90\,\% pada \qty{20}{\celsius}.
  \item Mengapa kulit yang hangat itu berguna?
  \item Mengapa suplemen vitamin D dibuat dengan menyinari sebuah sterol lalu menghangatkannya,
        dan bukan dengan sintesis klasik?
  \item Nyatakan hasilnya: waktu untuk mengonversi 90\,\% pravitamin D$_3$ pada suhu tubuh.
\end{enumerate}
\end{problem}
