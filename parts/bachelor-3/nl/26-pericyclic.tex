\section{Vier families}

\begin{definition}[Pericyclische reactie]\label{def:b3:pericyclic:pericyclic}
Een \emph{pericyclische reactie}\index{pericyclische reactie} is een
geconcerteerde reactie waarvan de overgangstoestand een cyclische
rangschikking van wisselwerkende orbitalen is, waarin alle bindingen
tegelijk rond de ring gevormd en verbroken worden.
\end{definition}

\begin{definition}[Families van pericyclische reacties]\label{def:b3:pericyclic:families}
Bij een \emph{elektrocyclische reactie}\index{elektrocyclische reactie}
sluit een geconjugeerd polyeen zich tot een ring door tussen zijn
uiteinden een $\sigma$-binding te vormen, of omgekeerd. Bij een
\emph{cycloadditie}\index{cycloadditie} verbinden twee $\pi$-systemen zich tot een ring door twee $\sigma$-bindingen te
vormen; een cycloadditie $[m + n]$ verbindt componenten
van $m$ en $n$ atomen (of elektronen). Bij een
\emph{sigmatrope omlegging}\index{sigmatrope omlegging} migreert een $\sigma$-binding langs een $\pi$-systeem; bij een verschuiving $[i, j]$ verhuist de binding van de atomen 1 en 1 naar de atomen $i$ en $j$
van de twee fragmenten. Een
\emph{cheletrope reactie}\index{cheletrope reactie} vormt of verbreekt
twee bindingen met hetzelfde atoom, zoals wanneer zwaveldioxide een
sulfoleen verlaat. Bij een \emph{ene-reactie}\index{ene-reactie} addeert
een alkeen met een allylisch waterstofatoom aan een $\pi$-binding, waarbij het waterstofatoom meeverhuist.
\end{definition}

\begin{definition}[Suprafaciaal en antarafaciaal]\label{def:b3:pericyclic:faciality}
Een component reageert \emph{suprafaciaal}\index{suprafaciaal} wanneer zijn
twee nieuwe bindingen zich vormen, of zijn twee oude breken, aan
hetzelfde vlak van zijn $\pi$-systeem (of met
retentie bij een $\sigma$-binding), en \emph{antarafaciaal}\index{antarafaciaal} wanneer
ze aan tegenovergestelde vlakken liggen (of met inversie).
\end{definition}

\section{Elektrocyclische reacties}

\begin{definition}[Conrotatorisch en disrotatorisch]\label{def:b3:pericyclic:rotation-modes}
Bij een elektrocyclische ringsluiting draaien de twee eindgroepen om de
bindingen die ze met de keten verbinden. De draaiing is
\emph{conrotatorisch}\index{conrotatorisch} wanneer beide in dezelfde zin
draaien (beide met de klok mee, gezien langs de keten), en
\emph{disrotatorisch}\index{disrotatorisch} wanneer ze in tegengestelde
zin draaien.
\end{definition}

\begin{theorem}[Elektrocyclische reacties]\label{thm:b3:pericyclic:electrocyclic}
Een thermische \omterm{def:b3:pericyclic:families}{elektrocyclische reactie} van een polyeen met $4n$ $\pi$-elektronen is
\omterm{def:b3:pericyclic:rotation-modes}{conrotatorisch}, en met $4n + 2$ $\pi$-elektronen \omterm{def:b3:pericyclic:rotation-modes}{disrotatorisch}.
\end{theorem}

\begin{proof}
De nieuwe $\sigma$-binding vormt zich uit de orbitalen p aan de uiteinden van de HOMO,
die bij de thermische reactie de elektronen met de hoogste energie
bevat; ze is alleen bindend als de lobben die naar elkaar toe draaien,
hetzelfde teken hebben. Volgens de formule van Hückel
(\cref{ch:b3:solid-state}) zijn de coëfficiënten van orbitaal $k$ van een keten van $N$ atomen
evenredig met $\sin(jk\pi/(N + 1))$, zodat de twee uiteinden, $j = 1$ en $j = N$, de
coëfficiënten $\sin(k\pi/(N + 1))$ en $\sin(Nk\pi/(N + 1)) = (-1)^{k+1}\sin(k\pi/(N + 1))$ hebben: hetzelfde teken voor oneven $k$,
tegengestelde tekens voor even $k$. Met $N$ atomen en $N$ elektronen is de HOMO
$k = N/2$. Voor $N = 4n$ is $k = 2n$ even: de bovenste lobben van de uiteinden hebben
tegengestelde tekens, en de twee uiteinden moeten dezelfde kant op
draaien, elk met zijn lob van het passende teken naar binnen:
\omterm{def:b3:pericyclic:rotation-modes}{conrotatorisch}. Voor $N = 4n + 2$ is $k = 2n + 1$ oneven: de bovenste lobben
hebben hetzelfde teken, en ze naar elkaar toe draaien is \omterm{def:b3:pericyclic:rotation-modes}{disrotatorisch}.
\end{proof}

De regel is stereospecifiek. (\textit{E},\textit{E})-hexa-2,4-dieen zou zich bij
verhitting \omterm{def:b3:pericyclic:rotation-modes}{conrotatorisch} sluiten tot \textit{trans}-3,4-dimethylcyclobuteen; in de praktijk is het
gespannen cyclobuteen degene die opengaat, en
\textit{cis}-3,4-dimethylcyclobuteen opent zich \omterm{def:b3:pericyclic:rotation-modes}{conrotatorisch} tot
uitsluitend (\textit{E},\textit{Z})-hexa-2,4-dieen. (\textit{E},\textit{Z},\textit{E})-octa-2,4,6-trieen sluit zich
\omterm{def:b3:pericyclic:rotation-modes}{disrotatorisch} tot \textit{cis}-5,6-dimethylcyclohexa-1,3-dieen.

\begin{proposition}[Fotochemische elektrocyclische reacties]\label{prop:b3:pericyclic:photo-electrocyclic}
Onder licht zijn de regels omgekeerd: $4n$ elektronen, \omterm{def:b3:pericyclic:rotation-modes}{disrotatorisch}; $4n + 2$, \omterm{def:b3:pericyclic:rotation-modes}{conrotatorisch}.
\end{proposition}

\begin{proof}
De absorptie van een foton promoveert een elektron van de HOMO $k = N/2$ naar de LUMO $k = N/2 + 1$
(\cref{ch:b3:photochemistry}), die nu de hoogste bezette orbitaal is en
de uiteinden bepaalt. Haar $k$ heeft de tegengestelde pariteit, zodat het relatieve teken van haar
coëfficiënten aan de uiteinden omkeert, en daarmee de draaizin.
\end{proof}

\begin{method}[De stereochemie van een elektrocyclische reactie voorspellen]\label{met:b3:pericyclic:predict-stereo}
\begin{enumerate}
  \item Tel de $\pi$-elektronen van het open polyeen ($4n$ of $4n + 2$); noteer warmte of
        licht.
  \item Lees de wijze af: thermisch $4n$ con, $4n + 2$ dis; fotochemisch omgekeerd.
  \item Teken de uiteinden met hun substituenten, in de conformatie die
        zich sluit (s-cis); draai ze over 90° volgens de gekozen wijze.
  \item Lees af of de twee substituenten aan hetzelfde vlak van de ring
        eindigen (\textit{cis}) of aan tegenovergestelde vlakken (\textit{trans}). Doorloop voor een
        ringopening dezelfde analyse in omgekeerde richting.
\end{enumerate}
\end{method}

\section{Orbitaalcorrelatiediagrammen}

Het oorspronkelijke argument van Woodward en Hoffmann bekijkt alle
orbitalen, niet alleen de HOMO. Langs een conrotatorische weg behoudt het
molecuul een as $C_2$; langs een
disrotatorische weg een \omterm{def:b3:point-groups:mirror}{spiegelvlak} $\sigma$. Elke orbitaal van de beginstof en van het
product is symmetrisch (S) of antisymmetrisch (A) ten opzichte van het
element dat behouden blijft.

\begin{definition}[Orbitaalcorrelatiediagram]\label{def:b3:pericyclic:orbital-correlation}
Een \emph{orbitaalcorrelatiediagram}\index{orbitaalcorrelatiediagram}
verbindt elke orbitaal van de beginstof met de orbitaal van het product
met dezelfde symmetrie, ten opzichte van de \omterm{def:b3:point-groups:operation}{symmetrie-elementen} die langs
de reactieweg behouden blijven, in volgorde van energie: de laagste S
met de laagste S, enzovoort.
\end{definition}

\begin{proposition}[Behoud van orbitaalsymmetrie]\label{prop:b3:pericyclic:correlation}
Als elke orbitaal die in de beginstof bezet is, correleert met een
orbitaal die in het product bezet is, is de thermische reactie
toegestaan; als een bezette orbitaal correleert met een lege, is ze
verboden, met een hoge barrière.
\end{proposition}

\begin{proof}[Beargumenteerd]
Langs de weg behoudt een orbitaal haar symmetrie en verandert haar
energie continu; orbitalen met dezelfde symmetrie kunnen elkaar niet
kruisen (ze mengen en stoten elkaar af), orbitalen met verschillende
symmetrie wel. Wanneer een dubbel bezette orbitaal van de beginstof een
antibindende orbitaal van het product moet worden, leidt de
grondtoestand van de beginstof naar een dubbel aangeslagen toestand van
het product, en de energie van de grondtoestand stijgt steil tot de
toestanden met dezelfde symmetrie laat op de weg mengen: de barrière is
hoog.
\end{proof}

\begin{method}[Een correlatiediagram opbouwen]\label{met:b3:pericyclic:correlation-diagram}
\begin{enumerate}
  \item Zoek de \omterm{def:b3:point-groups:operation}{symmetrie-elementen} die de gevormde en verbroken bindingen
        middendoor delen en langs de hele gekozen weg behouden blijven.
  \item Som de betrokken orbitalen van beginstof en product op, in
        volgorde van energie, en label elke orbitaal S of A voor elk
        element.
  \item Verbind de laagste orbitaal van elk symmetrietype aan de ene kant
        met de laagste van hetzelfde type aan de andere kant, dan de
        volgende, zonder ooit twee lijnen met dezelfde symmetrie te
        kruisen.
  \item Controleer de bezette orbitalen: allemaal naar bezette, toegestaan;
        één naar een lege, verboden.
\end{enumerate}
\end{method}

\begin{definition}[Toestandscorrelatiediagram]\label{def:b3:pericyclic:state-correlation}
Een \emph{toestandscorrelatiediagram}\index{toestandscorrelatiediagram}
verbindt de elektronische toestanden van beginstof en product, elk
opgebouwd uit de orbitaalconfiguraties en benoemd naar haar totale
symmetrie, langs de weg; toestanden met dezelfde symmetrie kruisen
elkaar niet.
\end{definition}

Voor de verboden disrotatorische weg correleert de grondtoestand van butadieen ($\psi_1^2\psi_2^2$)
met een dubbel aangeslagen toestand van cyclobuteen ($\sigma^2\pi^{*2}$), maar de eerste aangeslagen
toestand ($\psi_1^2\psi_2\psi_3$) correleert met de eerste aangeslagen toestand van cyclobuteen ($\sigma^2\pi\pi^*$): vanuit de
aangeslagen toestand loopt de disrotatorische weg bergaf. Dat is de
fotochemische omkering van \cref{prop:b3:pericyclic:photo-electrocyclic},
gezien op de toestanden.

\section{Cycloaddities}

\begin{theorem}[Cycloaddities]\label{thm:b3:pericyclic:cycloaddition}
Een thermische \omterm{def:b3:pericyclic:families}{cycloadditie} $[m + n]$ waarin beide componenten \omterm{def:b3:pericyclic:faciality}{suprafaciaal} reageren, is
toegestaan als $m + n = 4q + 2$ elektronen, en verboden als $m + n = 4q$; onder licht is de
regel omgekeerd.
\end{theorem}

\begin{proof}
Beide nieuwe bindingen vormen zich tussen de HOMO van de ene component en
de LUMO van de andere, die met hun lobben aan de uiteinden tegenover
elkaar staan; als beide \omterm{def:b3:pericyclic:faciality}{suprafaciaal} reageren, zijn beide bindingen
alleen bindend als de coëfficiënten aan de uiteinden van de
HOMO en van de LUMO hetzelfde relatieve teken hebben. Voor een component van $N$
elektronen op $N$ atomen heeft de HOMO $k = N/2$ en de LUMO $k = N/2 + 1$; coëfficiënten aan de uiteinden met
hetzelfde teken voor oneven $k$, tegengesteld voor even $k$. In het geval [4+2] hebben de HOMO van het dieen ($k = 2$)
en de LUMO van etheen ($k = 2$) beide tegengestelde tekens aan de uiteinden: ze passen. In het
geval [2+2] ontmoet de HOMO van het ene etheen ($k = 1$, gelijke tekens) de LUMO van het andere
($k = 2$, tegengesteld): één uiteinde bindend, één antibindend. In het algemeen passen de HOMO van de component $m$
en de LUMO van de component $n$ als $m/2$ en $n/2 + 1$ dezelfde
pariteit hebben, d.w.z.\ als $(m + n)/2$ oneven is: $m + n = 4q + 2$. Onder licht speelt de enkelvoudig bezette LUMO van
de aangeslagen component de rol van zijn HOMO, en wordt de
pariteitsvoorwaarde omgekeerd.
\end{proof}

Thermisch geven twee alkenen geen cyclobutaan in één geconcerteerde
stap; onder licht wel (\cref{ch:b3:photochemistry}). Ketenen zijn de
uitzondering die de regel bevestigt: hun $\pi^*$-orbitaal van C=O, loodrecht op die van C=C, laat ze
met een alkeen reageren in een geometrie [$\pi2_s$ + $\pi2_a$], met één component \omterm{def:b3:pericyclic:faciality}{suprafaciaal} en het
keteen \omterm{def:b3:pericyclic:faciality}{antarafaciaal}, en ze vormen thermisch cyclobutanonen.

\begin{definition}[1,3-Dipolaire cycloaddities]\label{def:b3:pericyclic:dipolar}
Een \emph{1,3-dipool}\index{1,3-dipool} is een $\pi$-systeem van drie atomen met vier
elektronen, geschreven met formele ladingen aan de uiteinden of in het
midden, zoals een
azide $\ce{R-N=N+=N-}$, een nitriloxide of een ozonmolecuul. Een
\emph{1,3-dipolaire
cycloadditie}\index{1,3-dipolaire cycloadditie} is zijn \omterm{def:b3:pericyclic:families}{cycloadditie}
[3+2], een reactie met zes elektronen, met een alkeen of alkyn, die een
vijfring geeft.
\end{definition}

De kopergekatalyseerde reactie van een azide met een eindstandig alkyn,
die één regio-isomeer van een 1,2,3-triazool geeft, is de betrouwbare
koppelingsreactie van de klikchemie. Ozonolyse begint met een
\omterm{def:b3:pericyclic:dipolar}{1,3-dipolaire cycloadditie} van ozon aan het alkeen.

\section{Sigmatrope omleggingen en de algemene regel}

\begin{proposition}[Waterstofverschuivingen]\label{prop:b3:pericyclic:sigmatropic-h}
Een thermische verschuiving $[1, j]$ van waterstof langs een polyeen is \omterm{def:b3:pericyclic:faciality}{suprafaciaal} toegestaan als de
overgangstoestand $4q + 2$ elektronen bevat ($[1,5]$) en moet \omterm{def:b3:pericyclic:faciality}{antarafaciaal} zijn als ze er $4q$ bevat
($[1,3]$, $[1,7]$).
\end{proposition}

\begin{proof}
Beschouw de overgangstoestand als een waterstofatoom (1s) dat zich
verplaatst tussen de uiteinden van een
radicaal van $j$ atomen met $j$ elektronen; het waterstofatoom overbrugt de twee uiteinden van de
SOMO, $k = (j + 1)/2$. Voor $j = 5$ is $k = 3$ oneven, hebben de uiteinden aan één vlak lobben met hetzelfde teken,
en kan de orbitaal 1s aan dat vlak met beide binden: \omterm{def:b3:pericyclic:faciality}{suprafaciaal}. Voor $j = 3$ en $j = 7$
zijn $k = 2$ en $4$ even, liggen de lobben met hetzelfde teken aan de twee uiteinden aan tegenovergestelde vlakken, en
moet het waterstofatoom van het ene vlak naar het andere overgaan:
\omterm{def:b3:pericyclic:faciality}{antarafaciaal}. Een verschuiving $[1,3]$ kan niet over zo'n korte keten reiken; een verschuiving $[1,7]$ wel, langs een
helixvormig polyeen.
\end{proof}

\begin{definition}[Omleggingen van Cope en Claisen]\label{def:b3:pericyclic:cope-claisen}
De \emph{cope-omlegging}\index{cope-omlegging} is de \omterm{def:b3:pericyclic:families}{sigmatrope omlegging} [3,3]
van een hexa-1,5-dieen; de
\emph{claisenomlegging}\index{claisenomlegging} die van een
allylvinylether, of van een allylarylether, tot een $\gamma,\delta$-onverzadigde carbonylverbinding of een \textit{ortho}-allylfenol.
\end{definition}

De verschuiving [3,3] is een volledig \omterm{def:b3:pericyclic:faciality}{suprafaciaal} proces met zes
elektronen, dat thermisch toegestaan is;
het verloopt via een stoelvormige overgangstoestand, zoals een

\begin{theorem}[Regels van Woodward-Hoffmann]\label{thm:b3:pericyclic:woodward-hoffmann}
Een thermische \omterm{def:b3:pericyclic:pericyclic}{pericyclische reactie} is toegestaan als het totale aantal componenten $(4q + 2)_s$ en
$(4r)_a$ oneven is; een fotochemische als het even is.
\end{theorem}

\begin{proof}[Gedeeltelijk bewijs]
Voor elke hierboven behandelde familie stemt de telling overeen met de
analyse van de grensorbitalen: een thermische \omterm{def:b3:pericyclic:families}{cycloadditie} [4+2] heeft de componenten $\pi4_s$ en $\pi2_s$, één component $(4q + 2)_s$,
oneven: toegestaan; de reactie [2+2] heeft $\pi2_s + \pi2_s$, twee, even: verboden, terwijl
$\pi2_s + \pi2_a$ er één heeft: toegestaan; de conrotatorische opening van cyclobuteen is $\sigma2_s + \pi2_a$, één:
toegestaan; de suprafaciale verschuiving $[1,5]$-H is $\sigma2_s + \pi4_s$, waarbij alleen $\sigma2_s$ telt: één,
toegestaan. Het algemene bewijs, geldig voor elke rangschikking van
componenten, hoort bij meer gevorderde cursussen.
\end{proof}

\begin{definition}[Aromatische overgangstoestanden en overgangstoestanden van Möbius]\label{def:b3:pericyclic:mobius}
Een \emph{aromatische overgangstoestand}\index{aromatische overgangstoestand}
is een cyclische rangschikking van orbitalen in een pericyclische
overgangstoestand die, zoals een hückelring, gestabiliseerd wordt als ze
$4q + 2$ elektronen bevat. Een
\emph{overgangstoestand van Möbius}\index{overgangstoestand van Mobius@overgangstoestand van Möbius}
heeft een oneven aantal tekenwisselingen tussen naburige orbitalen rond
de ring (één antarafaciale draaiing); ze wordt gestabiliseerd met $4q$ elektronen.
\end{definition}

\begin{proposition}[Aromatische overgangstoestanden]\label{prop:b3:pericyclic:aromatic-ts}
Een thermische \omterm{def:b3:pericyclic:pericyclic}{pericyclische reactie} is toegestaan als haar
overgangstoestand aromatisch is:
hückeltopologie met $4q + 2$ elektronen, of möbiustopologie met $4q$ elektronen.
\end{proposition}

\begin{proof}[Beargumenteerd]
De cyclische rangschikking van orbitalen in de overgangstoestand gedraagt
zich als een cyclisch geconjugeerd molecuul. Een hückelring heeft een
niet-ontaard laagste niveau met paren
erboven, gevuld met $4q + 2$ elektronen (deel van jaar~2); een ring met één
tekenomkering heeft al zijn niveaus in paren, gevuld met $4q$. Een gevulde schil
betekent stabilisatie, een \omterm{def:b3:pericyclic:mobius}{aromatische overgangstoestand} en een lage
barrière. Het aantal tekenwisselingen is het aantal antarafaciale
componenten, zodat deze regel gelijkwaardig is met de regel van
Woodward-Hoffmann.
\end{proof}

\begin{method}[Componenten tekenen en tellen]\label{met:b3:pericyclic:components}
\begin{enumerate}
  \item Markeer de gevormde en verbroken bindingen; elk $\pi$-systeem of elke $\sigma$-binding die verandert, is een
        component, gelabeld met zijn type en aantal elektronen ($\pi4$, $\pi2$, $\sigma2$, $\omega0$ voor
        een lege orbitaal).
  \item Kies s of a voor elke component uit de geometrie (hetzelfde vlak,
        of tegenovergestelde vlakken).
  \item Tel de componenten $(4q + 2)_s$ en $(4r)_a$: oneven, thermisch toegestaan; even,
        fotochemisch toegestaan.
  \item Controleer met het aantal elektronen rond de ring: hückel met $4q + 2$,
        möbius met $4q$.
\end{enumerate}
\end{method}

\begin{history}[Orbitaalsymmetrie]
In 1965 publiceerden Robert Woodward, die de raadselachtige
stereochemie van elektrocyclische stappen ontdekt had tijdens de synthese
van vitamine B$_{12}$, en Roald Hoffmann de regels van het behoud van de
orbitaalsymmetrie. Kenichi Fukui had in 1952 de grensorbitalen
ingevoerd; Fukui en Hoffmann deelden de Nobelprijs voor de Scheikunde van
1981. Woodward, die de prijs van 1965 voor zijn syntheses ontvangen had,
overleed in 1979.
\end{history}

\section{Probleem: zonlicht, huid en vitamine D}

\begin{problem}[{Weekendprobleem --- zonlicht, huid en vitamine D: de
fotochemische ringopening, de thermische waterstofverschuiving, de
nevenproducten en de tijd die het lichaam nodig heeft}]
\label{pb:b3:pericyclic:1}
Gegevens van het probleem: bij \qty{37}{\celsius} is de omzetting van previtamine D$_3$ in vitamine D$_3$
van de eerste orde met $k = \qty{2.3e-2}{h^{-1}}$ (de omgekeerde reactie wordt verwaarloosd); haar activeringsenergie is
\qty{85}{kJ/mol}.

\medskip\noindent\textbf{Deel I --- De ringopening.}
\begin{enumerate}
  \item Welke binding van 7-dehydrocholesterol breekt, en welk $\pi$-systeem ontstaat?
  \item Hoeveel elektronen nemen deel, en tot welke familie behoort de
        reactie?
  \item Waarom is er licht nodig?
  \item Is de fotochemische opening \omterm{def:b3:pericyclic:rotation-modes}{conrotatorisch} of \omterm{def:b3:pericyclic:rotation-modes}{disrotatorisch}?
        Verantwoord met de orbitalen.
  \item Waarom moet de nieuwe centrale dubbele binding van previtamine D$_3$ \textit{Z} zijn?
  \item Waarom verloopt ze niet thermisch, in het donker, bij lichaamstemperatuur?
\end{enumerate}

\medskip\noindent\textbf{Deel II --- De waterstofverschuiving.}
\begin{enumerate}[resume]
  \item Noem de atomen waartussen het waterstofatoom verhuist, en het type
        verschuiving.
  \item Hoeveel elektronen nemen deel aan de overgangstoestand?
  \item Tel de componenten: is de thermische reactie \omterm{def:b3:pericyclic:faciality}{suprafaciaal} of
        \omterm{def:b3:pericyclic:faciality}{antarafaciaal} toegestaan?
  \item Waarom is een antarafaciale verschuiving hier mogelijk, maar een verschuiving $[1,3]$
        niet?
  \item Is de overgangstoestand van het type Hückel of Möbius, en is ze
        aromatisch?
  \item Waarom heeft de verschuiving geen licht nodig?
\end{enumerate}

\medskip\noindent\textbf{Deel III --- Nevenproducten.}
\begin{enumerate}[resume]
  \item Previtamine D$_3$ kan onder licht opnieuw sluiten. Volgens welke wijze, en waarom kan dat een
        ringstereo-isomeer van 7-dehydrocholesterol (lumisterol) geven?
  \item Licht kan ook de centrale dubbele binding \textit{Z} omzetten in \textit{E} (tachysterol). Waarom
        kan tachysterol de verschuiving $[1,7]$-H niet ondergaan?
  \item Waarom levert langdurig zonlicht niet steeds meer vitamine D op?
  \item Waarom is de ringopening snel maar de waterstofverschuiving traag?
\end{enumerate}

\medskip\noindent\textbf{Deel IV --- Kinetiek bij lichaamstemperatuur.}
\begin{enumerate}[resume]
  \item Bereken de halveringstijd van previtamine D$_3$ bij \qty{37}{\celsius}.
  \item Bereken de fractie die in één dag omgezet wordt.
  \item Bereken de tijd om 90\,\% om te zetten.
  \item Bereken $k$ bij \qty{20}{\celsius}.
  \item Bereken de tijd om 90\,\% om te zetten bij \qty{20}{\celsius}.
  \item Waarom is het nuttig dat de huid warm is?
  \item Waarom maakt men vitamine-D-supplementen door een sterol te
        bestralen en te verwarmen, en niet door een klassieke synthese?
  \item Formuleer het resultaat: de tijd om 90\,\% van het previtamine D$_3$ bij lichaamstemperatuur om te zetten.
\end{enumerate}
\end{problem}
