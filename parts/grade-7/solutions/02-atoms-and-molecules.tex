\section*{Chapter \ref{ch:g7:atoms-and-molecules} --- Atoms and Molecules}

\begin{solution}{exo:g7:atoms-and-molecules:1}
An \omterm{def:g7:atoms-and-molecules:atom}{atom} is one of the extremely small particles from which all matter
is built (about a hundred kinds exist). A \omterm{def:g7:atoms-and-molecules:molecule}{molecule} is a group of \omterm{def:g7:atoms-and-molecules:atom}{atoms}
held tightly together, the smallest piece of a substance that is still
that substance.
\end{solution}

\begin{solution}{exo:g7:atoms-and-molecules:2}
\ce{H}, \ce{C}, \ce{N}, \ce{O}, \ce{Na}, \ce{Fe}.
\end{solution}

\begin{solution}{exo:g7:atoms-and-molecules:3}
1 carbon \omterm{def:g7:atoms-and-molecules:atom}{atom} and 2 oxygen \omterm{def:g7:atoms-and-molecules:atom}{atoms}: 3 \omterm{def:g7:atoms-and-molecules:atom}{atoms} in all.
\end{solution}

\begin{solution}{exo:g7:atoms-and-molecules:4}
Water, oxygen, methane, nitrogen.
\end{solution}

\begin{solution}{exo:g7:atoms-and-molecules:5}
\ce{NO2}.
\end{solution}

\begin{solution}{exo:g7:atoms-and-molecules:6}
\ce{O2} is one \omterm{def:g7:atoms-and-molecules:molecule}{molecule} made of two joined oxygen \omterm{def:g7:atoms-and-molecules:atom}{atoms}; \ce{2O} is two
separate oxygen \omterm{def:g7:atoms-and-molecules:atom}{atoms}. \ce{CO} is a \omterm{def:g7:atoms-and-molecules:molecule}{molecule} of one carbon and one
oxygen \omterm{def:g7:atoms-and-molecules:atom}{atom} (two capital letters, two \omterm{def:g7:atoms-and-molecules:symbol}{symbols}); \ce{Co} is one \omterm{def:g7:atoms-and-molecules:atom}{atom} of
cobalt (one \omterm{def:g7:atoms-and-molecules:symbol}{symbol}, capital then small letter).
\end{solution}

\begin{solution}{exo:g7:atoms-and-molecules:7}
\ce{3H2O}: $3\times 2 = 6$ hydrogen \omterm{def:g7:atoms-and-molecules:atom}{atoms} and $3\times 1 = 3$ oxygen
\omterm{def:g7:atoms-and-molecules:atom}{atoms}. \ce{5O2}: no hydrogen, $5\times 2 = 10$ oxygen \omterm{def:g7:atoms-and-molecules:atom}{atoms}.
\end{solution}

\begin{solution}{exo:g7:atoms-and-molecules:8}
Black is carbon, red is oxygen: one carbon, two oxygen, \ce{CO2},
carbon dioxide.
\end{solution}

\begin{solution}{exo:g7:atoms-and-molecules:9}
Dalton's water was \ce{HO}. The real \omterm{def:g7:atoms-and-molecules:molecule}{molecule}, \ce{H2O}, has one more
hydrogen \omterm{def:g7:atoms-and-molecules:atom}{atom}: Dalton's \omterm{def:g7:atoms-and-molecules:molecule}{molecule} was missing one \omterm{def:g7:atoms-and-molecules:atom}{atom}.
\end{solution}

\begin{solution}{exo:g7:atoms-and-molecules:10}
Pure water is not a \omterm{def:g6:pure-substances-and-mixtures:mixture}{mixture}: all its \omterm{def:g7:atoms-and-molecules:molecule}{molecules} are identical \ce{H2O}
\omterm{def:g7:atoms-and-molecules:molecule}{molecules}. \omterm{def:g4:air-a-mixture-of-gases:air}{Air} is a \omterm{def:g6:pure-substances-and-mixtures:mixture}{mixture}: it contains nitrogen \omterm{def:g7:atoms-and-molecules:molecule}{molecules}, oxygen
\omterm{def:g7:atoms-and-molecules:molecule}{molecules}, argon \omterm{def:g7:atoms-and-molecules:atom}{atoms} and a few carbon dioxide \omterm{def:g7:atoms-and-molecules:molecule}{molecules}.
\end{solution}

\begin{solution}{exo:g7:atoms-and-molecules:11}
Ten million \omterm{def:g7:atoms-and-molecules:atom}{atoms} make \qty{1}{mm}, so \qty{1}{cm} $= \qty{10}{mm}$
takes $10 \times$ ten million $=$ one hundred million \omterm{def:g7:atoms-and-molecules:atom}{atoms}.
$\qty{100}{m} = \num{100000}$~mm takes $\num{100000}\times$ ten million
$=$ one million million \omterm{def:g7:atoms-and-molecules:atom}{atoms}.
\end{solution}

\begin{solution}{exo:g7:atoms-and-molecules:12}
$6 + 12 + 6 = 24$ \omterm{def:g7:atoms-and-molecules:atom}{atoms}. Hydrogen: $\frac{12}{24} = \qty{50}{\percent}$.
Carbon: $\frac{6}{24} = \qty{25}{\percent}$ (and oxygen the other
\qty{25}{\percent}).
\end{solution}

\begin{solution}{pb:g7:atoms-and-molecules:1}
\textbf{1.} Nitrogen \ce{N2}, 2 \omterm{def:g7:atoms-and-molecules:atom}{atoms}; oxygen \ce{O2}, 2 \omterm{def:g7:atoms-and-molecules:atom}{atoms}; carbon
dioxide \ce{CO2}, 3 \omterm{def:g7:atoms-and-molecules:atom}{atoms}.

\textbf{2.} Argon \omterm{def:g7:atoms-and-molecules:atom}{atoms} do not join into \omterm{def:g7:atoms-and-molecules:molecule}{molecules}: argon is made of
single \omterm{def:g7:atoms-and-molecules:atom}{atoms}.

\textbf{3.} Nitrogen: $78 \times 2 = 156$ \omterm{def:g7:atoms-and-molecules:atom}{atoms}. Oxygen: $21\times 2 =
42$ \omterm{def:g7:atoms-and-molecules:atom}{atoms}.

\textbf{4.} $156 + 42 + 1 = 199$ \omterm{def:g7:atoms-and-molecules:atom}{atoms}.

\textbf{5.} $\frac{156}{199} \approx 0.78$, about \qty{78}{\percent}.

\textbf{6.} In oxygen \omterm{def:g7:atoms-and-molecules:molecule}{molecules}: $16\times 2 = 32$; in carbon dioxide
\omterm{def:g7:atoms-and-molecules:molecule}{molecules}: $4 \times 2 = 8$; in all $32 + 8 = 40$.

\textbf{7.} $42 - 40 = 2$ oxygen \omterm{def:g7:atoms-and-molecules:atom}{atoms} are missing: one oxygen \omterm{def:g7:atoms-and-molecules:molecule}{molecule}'s
worth. They have gone into water \omterm{def:g7:atoms-and-molecules:molecule}{molecules}, \ce{H2O}, which the body makes
with the hydrogen of its food; that water leaves partly as water vapour in
the breath (removed in these figures).

\textbf{8.} Breathed in: 0. Breathed out: 4 (one in each \ce{CO2}).

\textbf{9.} From the food the body uses: sugars and fats contain carbon
\omterm{def:g7:atoms-and-molecules:atom}{atoms}, and the body releases them in the carbon dioxide it breathes out.

\textbf{10.} $21 - 16 = 5$ oxygen \omterm{def:g7:atoms-and-molecules:molecule}{molecules} have disappeared and 4 carbon
dioxide \omterm{def:g7:atoms-and-molecules:molecule}{molecules} have appeared.

\textbf{11.} Nitrogen: two blue balls joined by three sticks. Oxygen: two
red balls joined by two sticks. Carbon dioxide: a black ball between two
red balls, each joined to it by two sticks, in a straight line. Argon: a
single pale blue ball.

\textbf{12.} $156 + 32 + 12 + 1 = 201$ \omterm{def:g7:atoms-and-molecules:atom}{atoms} (the 4 carbon dioxide
\omterm{def:g7:atoms-and-molecules:molecule}{molecules} hold 12 \omterm{def:g7:atoms-and-molecules:atom}{atoms}). That is $201 - 199 = 2$ \omterm{def:g7:atoms-and-molecules:atom}{atoms} more than the \omterm{def:g4:air-a-mixture-of-gases:air}{air}
breathed in: 4 carbon \omterm{def:g7:atoms-and-molecules:atom}{atoms} gained from food, minus the 2 oxygen \omterm{def:g7:atoms-and-molecules:atom}{atoms} that
left in water.
\end{solution}
