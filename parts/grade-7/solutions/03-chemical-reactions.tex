\section*{Chapter \ref{ch:g7:chemical-reactions} --- Chemical Reactions: Reactants and Products}

\begin{solution}{exo:g7:chemical-reactions:1}
Physical: ice melting, sugar dissolving. Chemical: a match \omterm{def:g4:what-a-fire-needs:burning}{burning},
bread toasting, a nail rusting.
\end{solution}

\begin{solution}{exo:g7:chemical-reactions:2}
The \omterm{def:g7:chemical-reactions:reactant}{reactants} are the species used up; the \omterm{def:g7:chemical-reactions:reactant}{products} are the new species
formed.
\end{solution}

\begin{solution}{exo:g7:chemical-reactions:3}
carbon $+$ oxygen $\longrightarrow$ carbon dioxide.
\end{solution}

\begin{solution}{exo:g7:chemical-reactions:4}
\omterm{def:g7:chemical-reactions:reactant}{Reactants}: iron and oxygen. \omterm{def:g7:chemical-reactions:reactant}{Product}: iron oxide.
\end{solution}

\begin{solution}{exo:g7:chemical-reactions:5}
The mist shows water; the milky limewater shows carbon dioxide.
\end{solution}

\begin{solution}{exo:g7:chemical-reactions:6}
zinc $+$ hydrochloric acid $\longrightarrow$ zinc chloride $+$ hydrogen.
\end{solution}

\begin{solution}{exo:g7:chemical-reactions:7}
Before: 4 (in the methane \omterm{def:g7:atoms-and-molecules:molecule}{molecule}). After: $2 + 2 = 4$ (two in each
water \omterm{def:g7:atoms-and-molecules:molecule}{molecule}).
\end{solution}

\begin{solution}{exo:g7:chemical-reactions:8}
No new species appears: the salt is still there, spread through the
water, and it is got back by \omterm{def:g3:separating-mixtures:evaporate}{evaporating} the water. It is a \omterm{def:g7:chemical-reactions:physical-transformation}{physical
transformation}.
\end{solution}

\begin{solution}{exo:g7:chemical-reactions:9}
Most of the wood has reacted with oxygen to give \omterm{def:g7:chemical-reactions:reactant}{products} that are
gases, carbon dioxide and water vapour, which leave into the \omterm{def:g4:air-a-mixture-of-gases:air}{air}. Only
the ash stays.
\end{solution}

\begin{solution}{exo:g7:chemical-reactions:10}
hydrogen $+$ oxygen $\longrightarrow$ water. The \omterm{def:g7:chemical-reactions:reactant}{product} turns anhydrous
copper sulfate blue.
\end{solution}

\begin{solution}{exo:g7:chemical-reactions:11}
Before: $2 \times 2 = 4$ hydrogen \omterm{def:g7:atoms-and-molecules:atom}{atoms} and 2 oxygen \omterm{def:g7:atoms-and-molecules:atom}{atoms}. After: two
water \omterm{def:g7:atoms-and-molecules:molecule}{molecules} hold $2 \times 2 = 4$ hydrogen \omterm{def:g7:atoms-and-molecules:atom}{atoms} and $2 \times 1 =
2$ oxygen \omterm{def:g7:atoms-and-molecules:atom}{atoms}. The same \omterm{def:g7:atoms-and-molecules:atom}{atoms}, in the same numbers.
\end{solution}

\begin{solution}{exo:g7:chemical-reactions:12}
An oxygen \omterm{def:g7:atoms-and-molecules:atom}{atom} cannot come from nowhere: the drawing has 2 oxygen \omterm{def:g7:atoms-and-molecules:atom}{atoms}
before and 3 after. Correct: one carbon \omterm{def:g7:atoms-and-molecules:atom}{atom} $+$ one oxygen \omterm{def:g7:atoms-and-molecules:molecule}{molecule}
$\longrightarrow$ one carbon dioxide \omterm{def:g7:atoms-and-molecules:molecule}{molecule}, with nothing left over.
\end{solution}

\begin{solution}{pb:g7:chemical-reactions:1}
\textbf{1.} Methane and oxygen.

\textbf{2.} Water.

\textbf{3.} Carbon dioxide.

\textbf{4.} No: it is not used up, and it is not written in the \omterm{def:g7:chemical-reactions:word-equation}{word
equation}.

\textbf{5.} methane $+$ oxygen $\longrightarrow$ carbon dioxide $+$
water.

\textbf{6.} Chemical: two species disappear (methane, oxygen), and two
new species appear, detected by tests; and heat and light are given out.

\textbf{7.} \ce{CH4}, \ce{O2}, \ce{CO2}, \ce{H2O}.

\textbf{8.} 1 carbon, 4 hydrogen, $2 \times 2 = 4$ oxygen.

\textbf{9.} 1 carbon; $2 \times 2 = 4$ hydrogen; $2 + 2 \times 1 = 4$
oxygen. The same \omterm{def:g7:atoms-and-molecules:atom}{atoms} in the same numbers: they have only been
rearranged.

\textbf{10.} $10 \times 2 = 20$ oxygen \omterm{def:g7:atoms-and-molecules:molecule}{molecules}.

\textbf{11.} 10 carbon dioxide \omterm{def:g7:atoms-and-molecules:molecule}{molecules}.

\textbf{12.} $10 \times 2 = 20$ water \omterm{def:g7:atoms-and-molecules:molecule}{molecules}.
\end{solution}
